   112→
   113→  // Apply source filter client-side on fetched molecules, then sort

   121→    else if (sortBy === 'molarMass') filtered.sort((a, b) => b.molarMass - a.molarMass)
   122→    else if (sortBy === 'risk') {
   123→      const riskOrder: Record<string, number> = { critical: 0, high: 1, moderate: 2, low: 3 }
   124→      filtered.sort((a, b) => (riskOrder[a.riskLevel] ?? 4) - (riskOrder[b.riskLevel] ?? 4))

   128→

   132→    apiMolecules.forEach((mol) => { counts[mol.riskLevel] = (counts[mol.riskLevel] || 0) + 1 })
   133→    return counts
   134→  })()

   144→        const data = await res.json()

   146→      }
   147→    } catch { /* ignore */ }
   148→    setDegradationLoading(false)
   149→  }
   150→
   151→  const handleAddDegradation = async () => {

   180→        toast({ title: 'Error', description: err.error || 'Failed to add degradation product', variant: 'destructive' })

   182→    } catch {
   183→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })

   189→  const handleImportCSV = async (e: React.ChangeEvent<HTMLInputElement>) => {
   190→    const file = e.target.files?.[0]
   191→    if (!file) return
   192→    setImporting(true)
   193→    try {
   194→      const text = await file.text()
   195→      const lines = text.split(/\r?\n/).filter((l) => l.trim())
   196→      if (lines.length < 2) {
   197→        toast({ title: 'Import failed', description: 'CSV must have a header row and at least one data row', variant: 'destructive' })
   198→        return
   199→      }
   200→      const parseCSVLine = (line: string): string[] => {
   201→        const out: string[] = []
   202→        let cur = ''
   203→        let inQ = false
   204→        for (let i = 0; i < line.length; i++) {
   205→          const ch = line[i]
   206→          if (ch === '"' && line[i + 1] === '"') { cur += '"'; i++ }
   207→          else if (ch === '"') { inQ = !inQ }
   208→          else if (ch === ',' && !inQ) { out.push(cur); cur = '' }
   209→          else { cur += ch }
   210→        }
   211→        out.push(cur)
   212→        return out
   213→      }
   214→      const headers = parseCSVLine(lines[0]).map((h) => h.trim().toLowerCase())
   215→      const nameIdx = headers.findIndex((h) => h === 'name')
   216→      if (nameIdx === -1) {
   217→        toast({ title: 'Import failed', description: 'CSV must have a "name" column', variant: 'destructive' })
   218→        return
   219→      }
   220→      const casIdx = headers.findIndex((h) => h === 'casnumber' || h === 'cas')
   221→      const formulaIdx = headers.findIndex((h) => h === 'formula')
   222→      const smilesIdx = headers.findIndex((h) => h === 'smiles')
   223→      const mwIdx = headers.findIndex((h) => h === 'molarmass' || h === 'mw')
   224→      const logPIdx = headers.findIndex((h) => h === 'logp')
   225→      const scoreIdx = headers.findIndex((h) => h === 'stabilityscore' || h === 'predictedstabilityscore' || h === 'score')
   226→      const riskIdx = headers.findIndex((h) => h === 'risklevel' || h === 'risk')
   227→
   228→      let success = 0, failed = 0
   229→      for (let i = 1; i < lines.length; i++) {
   230→        const cells = parseCSVLine(lines[i])
   231→        const name = cells[nameIdx]?.trim()
   232→        if (!name) { failed++; continue }
   233→        try {
   234→          const res = await fetch('/api/molecules', {
   235→            method: 'POST',
   236→            headers: { 'Content-Type': 'application/json' },
   237→            body: JSON.stringify({
   238→              name,
   239→              casNumber: casIdx >= 0 ? cells[casIdx]?.trim() || undefined : undefined,
   240→              formula: formulaIdx >= 0 ? cells[formulaIdx]?.trim() || undefined : undefined,
   241→              smiles: smilesIdx >= 0 ? cells[smilesIdx]?.trim() || undefined : undefined,
   242→              molarMass: mwIdx >= 0 ? parseFloat(cells[mwIdx]) || undefined : undefined,
   243→              logP: logPIdx >= 0 ? parseFloat(cells[logPIdx]) || undefined : undefined,
   244→              predictedStabilityScore: scoreIdx >= 0 ? parseFloat(cells[scoreIdx]) || undefined : undefined,
   245→              riskLevel: riskIdx >= 0 ? cells[riskIdx]?.trim().toLowerCase() || 'low' : 'low',
   246→            }),
   247→          })
   248→          if (res.ok) success++; else failed++
   249→        } catch { failed++ }
   250→      }
   251→      toast({
   252→        title: 'Import complete',
   253→        description: `Imported ${success} molecule${success !== 1 ? 's' : ''}${failed ? `, ${failed} failed` : ''}`,
   254→        variant: failed > 0 ? 'destructive' : 'default',
   255→      })
   256→      handleRefresh()
   257→    } catch {
   258→      toast({ title: 'Import failed', description: 'Failed to read file', variant: 'destructive' })
   259→    } finally {
   260→      setImporting(false)
   261→      if (fileInputRef.current) fileInputRef.current.value = ''
   262→    }
   263→  }

   296→          dataSource: newMolecule.dataSource,
   297→          description: newMolecule.description || undefined,

   306→        const errBody = await res.json().catch(() => ({}))
   307→        toast({ title: 'Error', description: errBody.error || 'Failed to create molecule', variant: 'destructive' })
   308→      }
   309→    } catch {
   310→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   311→    } finally {
   312→      setCreating(false)
   313→    }

   328→          molarMass: m.molarMass ?? '',

   331→          riskLevel: m.riskLevel ?? '',
   332→          dataSource: m.dataSource ?? '',
   333→          meltingPoint: m.meltingPoint ?? '',
   334→          boilingPoint: m.boilingPoint ?? '',
   335→          description: m.description ?? '',
   336→        }))
   337→        if (!rows.length) {

   339→        } else {

   342→        }
   343→      } else {
   344→        toast({ title: 'Export failed', description: 'Failed to fetch molecules', variant: 'destructive' })
   345→      }
   346→    } catch {
   347→      toast({ title: 'Export failed', description: 'Network error', variant: 'destructive' })
   348→    } finally {
   349→      setExporting(false)
   350→    }
   351→  }
   352→

   354→    <motion.div
   355→      initial={{ opacity: 0, y: 20 }}

   357→      transition={{ duration: 0.4 }}
   358→      className="space-y-6"
   359→    >
   360→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   361→        <div>

   363→          <p className="text-muted-foreground">Browse and search chemical compounds with stability assessments</p>

   381→          />

   417→            size="sm"
   418→            className={`h-7 text-xs rounded-full ${riskFilter === level ? (RISK_PILL_ACTIVE[level] || 'bg-emerald-600 text-white') : ''} ${RISK_PILL_OUTLINE[level] || ''}`}
   419→            onClick={() => { setRiskFilter(level); setCurrentPageNum(1) }}
   420→          >
   421→            {level === 'all' ? 'All' : level.charAt(0).toUpperCase() + level.slice(1)}

   427→      {/* Search & Filters */}

   434→            onChange={(e) => { setSearchQuery(e.target.value); setCurrentPageNum(1) }}
   435→            className="pl-9"
   436→          />

   442→          <SelectContent>

   444→            <SelectItem value="stability">Stability</SelectItem>
   445→            <SelectItem value="molarMass">Molar Mass</SelectItem>
   446→            <SelectItem value="risk">Risk Level</SelectItem>
   447→          </SelectContent>
   448→        </Select>
   449→        <Select value={riskFilter} onValueChange={(v) => { setRiskFilter(v); setCurrentPageNum(1) }}>
   450→          <SelectTrigger className="w-[140px]">
   451→            <SelectValue placeholder="Risk Level" />
   452→          </SelectTrigger>
   453→          <SelectContent>
   454→            <SelectItem value="all">All Risks</SelectItem>
   455→            <SelectItem value="low">Low</SelectItem>
   456→            <SelectItem value="moderate">Moderate</SelectItem>
   457→            <SelectItem value="high">High</SelectItem>
   458→            <SelectItem value="critical">Critical</SelectItem>
   459→          </SelectContent>
   460→        </Select>
   461→        <Select value={sourceFilter} onValueChange={(v) => { setSourceFilter(v); setCurrentPageNum(1) }}>
   462→          <SelectTrigger className="w-[140px]">
   463→            <SelectValue placeholder="Data Source" />
   464→          </SelectTrigger>
   465→          <SelectContent>
   466→            <SelectItem value="all">All Sources</SelectItem>
   467→            <SelectItem value="PubChem">PubChem</SelectItem>
   468→            <SelectItem value="ChEMBL">ChEMBL</SelectItem>
   469→            <SelectItem value="Manual">Manual</SelectItem>
   470→          </SelectContent>
   471→        </Select>
   472→      </div>
   473→
   474→      {/* Results count */}
   475→      <p className="text-sm text-muted-foreground">
   476→        {loading ? 'Loading...' : `Showing ${displayed.length} of ${totalCount} molecules`}
   477→      </p>
   478→
   479→      {/* Table / Grid View */}
   480→      {viewMode === 'table' ? (
   481→      <Card className="backdrop-blur-sm bg-card/80">
   482→        <CardContent className="p-0">
   483→          <div className="max-h-96 overflow-y-auto">
   484→            {loading ? (
   485→              <div className="space-y-2 p-4">
