   132→   132→function getScoreColor(score: number): string {

   139→   139→function getScoreLabel(score: number): string {

   157→   157→function ScoreRing({ score, size = 200 }: { score: number; size?: number }) {
   158→   158→  const strokeWidth = 14
   159→   159→  const radius = (size - strokeWidth) / 2
   160→   160→  const circumference = 2 * Math.PI * radius
   161→   161→  const offset = circumference - (score / 100) * circumference
   162→   162→  const color = getScoreColor(score)
   163→   163→  const label = getScoreLabel(score)
   164→   164→
   165→   165→  return (
   166→   166→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   167→   167→      <svg width={size} height={size} className="-rotate-90">
   168→   168→        <defs>
   169→   169→          <linearGradient id="scoreGradient" x1="0%" y1="0%" x2="100%" y2="100%">
   170→   170→            <stop offset="0%" stopColor={color} stopOpacity="0.8" />
   171→   171→            <stop offset="100%" stopColor={color} stopOpacity="1" />
   172→   172→          </linearGradient>
   173→   173→        </defs>
   174→   174→        {/* Track */}
   175→   175→        <circle
   176→   176→          cx={size / 2}
   177→   177→          cy={size / 2}
   178→   178→          r={radius}
   179→   179→          fill="none"
   180→   180→          strokeWidth={strokeWidth}
   181→   181→          className="stroke-muted/30"
