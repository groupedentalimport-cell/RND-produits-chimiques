   237→export function exportCSV(data: Record<string, unknown>[], filename: string) {
   238→  if (!data.length) return
   239→  const headers = Object.keys(data[0])
   240→  const csvLines = [headers.join(',')]
   241→  for (const row of data) {
   242→    const values = headers.map((h) => {
   243→      const v = row[h] ?? ''
   244→      const s = String(v).replace(/"/g, '""')
   245→      return /[",\n]/.test(s) ? `"${s}"` : s
   246→    })
   247→    csvLines.push(values.join(','))
   248→  }
   249→  const blob = new Blob([csvLines.join('\n')], { type: 'text/csv;charset=utf-8;' })
   250→  const url = URL.createObjectURL(blob)
   251→  const a = document.createElement('a')
   252→  a.href = url
   253→  a.download = filename
   254→  document.body.appendChild(a)
   255→  a.click()
   256→  document.body.removeChild(a)
   257→  URL.revokeObjectURL(url)
   258→}

   260→// Format a number for CSV output (null -> empty string)
   261→export function fmtNum(v: number | null | undefined): string {
   262→  return v === null || v === undefined ? '' : String(v)

   265→// ── Chemical formula subscript formatter ──────────────────────────────
   266→// Converts "C9H8O4" → "C₉H₈O₄", "H2O2" → "H₂O₂", "NaCl" stays "NaCl"
   267→export const SUBSCRIPT_DIGITS: Record<string, string> = {
   268→  '0': '₀', '1': '₁', '2': '₂', '3': '₃', '4': '₄',
   269→  '5': '₅', '6': '₆', '7': '₇', '8': '₈', '9': '₉',
   270→}
   271→export function formatFormula(formula: string | null | undefined): string {
   272→  if (!formula) return '—'
   273→  // If already contains unicode subscripts, leave as-is
   274→  if (/[₀-₉]/.test(formula)) return formula
   275→  // Replace any digit that follows a letter (or another digit) with subscript
   276→  return formula.replace(/(\d+)/g, (m) =>
   277→    m.split('').map((d) => SUBSCRIPT_DIGITS[d] ?? d).join('')
   278→  )
   279→}

   324→// Helper: format an ISO timestamp into a relative-time string ("2h ago", "Yesterday", "3d ago")
   325→export function formatRelativeTime(timestamp: string): string {
   326→  const now = Date.now()
   327→  const then = new Date(timestamp).getTime()
   328→  if (Number.isNaN(then)) return ''
   329→  const diffMs = now - then
   330→  const sec = Math.max(0, Math.floor(diffMs / 1000))
   331→  if (sec < 60) return 'just now'
   332→  const min = Math.floor(sec / 60)
   333→  if (min < 60) return `${min}m ago`
   334→  const hr = Math.floor(min / 60)
   335→  if (hr < 24) return `${hr}h ago`
   336→  const day = Math.floor(hr / 24)
   337→  if (day === 1) return 'Yesterday'
   338→  if (day < 7) return `${day}d ago`
   339→  const wk = Math.floor(day / 7)
   340→  if (wk < 5) return `${wk}w ago`
   341→  const mo = Math.floor(day / 30)
   342→  if (mo < 12) return `${mo}mo ago`
   343→  return `${Math.floor(day / 365)}y ago`
   344→}
