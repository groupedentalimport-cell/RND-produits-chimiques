    27→  concentration: number;

    29→  degradation_percent: number;
    30→  is_oos: boolean;

    33→
    34→interface SimulationResult {
    35→  condition_code: string;
    36→  condition_description: string;
    37→  substance_name: string;
    38→  initial_concentration: number;
    39→  concentration_unit: string;
    40→  temperature_c: number;
    41→  humidity_percent: number | null;
    42→  kinetic_order: number;
    43→  activation_energy: number;
    44→  rate_constant: number;
    45→  time_points: TimePoint[];
    46→  shelf_life_days: number | null;
    47→  shelf_life_months: number | null;
    48→  t90_days: number | null;
    49→  t95_days: number | null;
    50→  t99_days: number | null;
    51→  regression: {

    54→    r_squared: number | null;
    55→    ci_lower_days: number | null;
    56→    ci_upper_days: number | null;
    57→  };
    58→  simulation_type: string;
    59→  ich_reference: string;
    60→  computed_at: string;
    61→}
    62→
    63→interface ProtocolResult {
    64→  climate_zone: string;
    65→  conditions_simulated: string[];
    66→  simulations: Record<string, SimulationResult>;
    67→  summary: {
    68→    conditions: number;
    69→    shelf_lives: Record<string, { shelf_life_days: number | null; shelf_life_months: number | null }>;
    70→    critical_findings: string[];
    71→  };
    72→}
    73→
    74→interface MonteCarloResult {
    75→  n_simulations: number;
    76→  mean_shelf_life_days: number;
    77→  std_shelf_life_days: number;
    78→  mean_shelf_life_months: number;
    79→  confidence_level: number;
    80→  ci_lower_days: number;
    81→  ci_upper_days: number;
    82→  ci_lower_months: number;
    83→  ci_upper_months: number;
    84→  min_days: number;
    85→  max_days: number;
    86→  histogram: { bins: number[]; counts: number[] };
    87→}
    88→
    89→interface MolecularRisk {
    90→  overall_stability_score: number;
    91→  functional_groups: Array<{
    92→    name: string;
    93→    count: number;
    94→    risk: string;
    95→    weight: number;
    96→    description: string;
    97→  }>;
    98→  pathway_risks: Record<string, number>;
    99→  recommendations: string[];

   133→// ── Main Component ────────────────────────────────────────────────────
   134→
   135→const StabilitySimulator: React.FC = () => {
   136→  const [activeTab, setActiveTab] = useState<Tab>("simulate");
   137→  const [loading, setLoading] = useState(false);
   138→  const [error, setError] = useState<string | null>(null);
   139→
   140→  // Simulation state
   141→  const [simResult, setSimResult] = useState<SimulationResult | null>(null);
   142→  const [protocolResult, setProtocolResult] = useState<ProtocolResult | null>(null);
   143→  const [mcResult, setMcResult] = useState<MonteCarloResult | null>(null);
   144→  const [riskResult, setRiskResult] = useState<MolecularRisk | null>(null);
   145→  const [extrapResult, setExtrapResult] = useState<any>(null);
   146→
   147→  // Form state
   148→  const [form, setForm] = useState({
   149→    substance_name: "Acetylsalicylic Acid",
   150→    initial_concentration: 100,
   151→    concentration_unit: "mg/mL",
   152→    temperature_c: 25,
   153→    humidity_percent: 60,
   154→    activation_energy: 75000,
   155→    pre_exponential_factor: 1e10,
   156→    kinetic_order: 1,
   157→    duration_months: 36,
   158→    spec_lower: 90,
   159→    climate_zone: "II",
   160→    include_stress: true,
   161→    include_photostability: false,
   162→  });
   163→
   164→  const [extrapForm, setExtrapForm] = useState({
   165→    accelerated_rate_constant: 0.005,
   166→    accelerated_temperature_c: 40,
   167→    storage_temperature_c: 25,
   168→    activation_energy: 75000,
   169→    kinetic_order: 1,
   170→  });
   171→
   172→  const [mcForm, setMcForm] = useState({
   173→    mean_activation_energy: 75000,
   174→    std_activation_energy: 7500,
   175→    mean_pre_exponential: 1e10,
   176→    std_pre_exponential: 1e9,
   177→    temperature_c: 25,
   178→    kinetic_order: 1,
   179→    n_simulations: 5000,
   180→    confidence_level: 0.95,
   181→  });
   182→
   183→  const [smilesInput, setSmilesInput] = useState("CC(=O)Oc1ccccc1C(=O)O");
   184→
   185→  // ── API calls ─────────────────────────────────────────────────────
   186→
   187→  const runSimulation = useCallback(async () => {
   188→    setLoading(true);
   189→    setError(null);
   190→    try {
   191→      const res = await fetch(`${API_BASE}/simulate`, {
   192→        method: "POST",
   193→        headers: { "Content-Type": "application/json" },
   194→        body: JSON.stringify(form),
   195→      });
   196→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   197→      const data = await res.json();
   198→      setSimResult(data.simulation);
   199→    } catch (e: any) {
   200→      setError(e.message);
   201→    } finally {
   202→      setLoading(false);
   203→    }
   204→  }, [form]);
   205→
   206→  const runProtocol = useCallback(async () => {
   207→    setLoading(true);
   208→    setError(null);
   209→    try {
   210→      const res = await fetch(`${API_BASE}/protocol`, {
   211→        method: "POST",
   212→        headers: { "Content-Type": "application/json" },
   213→        body: JSON.stringify({
   214→          substance_name: form.substance_name,
   215→          initial_concentration: form.initial_concentration,
   216→          concentration_unit: form.concentration_unit,
   217→          activation_energy: form.activation_energy,
   218→          pre_exponential_factor: form.pre_exponential_factor,
   219→          kinetic_order: form.kinetic_order,
   220→          climate_zone: form.climate_zone,
   221→          spec_lower: form.spec_lower,
   222→          include_stress: form.include_stress,
   223→          include_photostability: form.include_photostability,
   224→        }),
   225→      });
   226→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   227→      const data = await res.json();
   228→      setProtocolResult(data);
   229→    } catch (e: any) {
   230→      setError(e.message);
   231→    } finally {
   232→      setLoading(false);
   233→    }
   234→  }, [form]);
   235→
   236→  const runExtrapolation = useCallback(async () => {
   237→    setLoading(true);
   238→    setError(null);
   239→    try {
   240→      const res = await fetch(`${API_BASE}/extrapolate`, {
   241→        method: "POST",
   242→        headers: { "Content-Type": "application/json" },
   243→        body: JSON.stringify(extrapForm),
   244→      });
   245→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   246→      const data = await res.json();
   247→      setExtrapResult(data.extrapolation);
   248→    } catch (e: any) {
   249→      setError(e.message);
   250→    } finally {
   251→      setLoading(false);
   252→    }
   253→  }, [extrapForm]);
   254→
   255→  const runMonteCarlo = useCallback(async () => {
   256→    setLoading(true);
   257→    setError(null);
   258→    try {
   259→      const res = await fetch(`${API_BASE}/monte-carlo`, {
   260→        method: "POST",
   261→        headers: { "Content-Type": "application/json" },
   262→        body: JSON.stringify(mcForm),
   263→      });
   264→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   265→      const data = await res.json();
   266→      setMcResult(data.monte_carlo);
   267→    } catch (e: any) {
   268→      setError(e.message);
   269→    } finally {
   270→      setLoading(false);
   271→    }
   272→  }, [mcForm]);
   273→
   274→  const runMolecularRisk = useCallback(async () => {
   275→    setLoading(true);
   276→    setError(null);
   277→    try {
   278→      const res = await fetch(`${API_BASE}/molecular-risk`, {
   279→        method: "POST",
   280→        headers: { "Content-Type": "application/json" },
   281→        body: JSON.stringify({ smiles: smilesInput }),
   282→      });
   283→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   284→      const data = await res.json();
   285→      setRiskResult(data.risk_assessment);
   286→    } catch (e: any) {
   287→      setError(e.message);
   288→    } finally {
   289→      setLoading(false);
   290→    }
   291→  }, [smilesInput]);
   292→
   293→  // ── Chart data ────────────────────────────────────────────────────
   294→
   295→  const simChartData = useMemo(() => {
   296→    if (!simResult) return [];
   297→    return simResult.time_points.map((tp) => ({
   298→      time: tp.time_months,
   299→      remaining: tp.percent_remaining,
   300→      degradation: tp.degradation_percent,
   301→      oos: tp.is_oos,
   302→    }));
   303→  }, [simResult]);
   304→
   305→  const protocolChartData = useMemo(() => {
   306→    if (!protocolResult) return [];
   307→    // Merge all time series into one chart-friendly format
   308→    const allTimes = new Set<number>();
   309→    Object.values(protocolResult.simulations).forEach((sim) => {
   310→      sim.time_points.forEach((tp) => allTimes.add(tp.time_months));
   311→    });
   312→    const sortedTimes = Array.from(allTimes).sort((a, b) => a - b);
   313→
   314→    return sortedTimes.map((t) => {
   315→      const point: any = { time: t };
   316→      Object.entries(protocolResult.simulations).forEach(([code, sim]) => {
   317→        const tp = sim.time_points.find((p) => p.time_months === t);
   318→        point[code] = tp ? tp.percent_remaining : null;
   319→      });
   320→      return point;
   321→    });
   322→  }, [protocolResult]);
   323→
   324→  const mcHistogramData = useMemo(() => {
   325→    if (!mcResult?.histogram) return [];
   326→    return mcResult.histogram.bins.slice(0, -1).map((bin, i) => ({
   327→      bin: Math.round(bin),
   328→      count: mcResult.histogram.counts[i],
   329→    }));
   330→  }, [mcResult]);
   331→
   332→  // ── Tabs ──────────────────────────────────────────────────────────
   333→
   334→  const tabs: { key: Tab; label: string; icon: string }[] = [
   335→    { key: "simulate", label: "Simulation", icon: "🧪" },
   336→    { key: "protocol", label: "Protocole ICH", icon: "📋" },
   337→    { key: "extrapolate", label: "Extrapolation", icon: "📈" },
   338→    { key: "monte-carlo", label: "Monte Carlo", icon: "🎲" },
   339→    { key: "molecular-risk", label: "Risque Moléculaire", icon: "⚗️" },
   340→  ];
   341→
   342→  return (
   343→    <div className="min-h-screen bg-gray-50 p-6">
   344→      {/* Header */}
   345→      <div className="mb-8">
   346→        <h1 className="text-3xl font-bold text-gray-900">
   347→          ⚗️ Simulateur de Stabilité Moléculaire
   348→        </h1>
   349→        <p className="text-gray-600 mt-2">
   350→          Études de stabilité conformes ICH Q1A-Q1F — Simulation cinétique dans le facteur temps
   351→        </p>
   352→      </div>
   353→
   354→      {/* Tabs */}
   355→      <div className="flex space-x-1 mb-6 bg-white rounded-lg p-1 shadow-sm">
   356→        {tabs.map((tab) => (
   357→          <button
   358→            key={tab.key}
   359→            onClick={() => setActiveTab(tab.key)}
   360→            className={`flex-1 py-2 px-4 rounded-md text-sm font-medium transition-colors ${
   361→              activeTab === tab.key
   362→                ? "bg-blue-600 text-white"
   363→                : "text-gray-600 hover:bg-gray-100"
   364→            }`}
   365→          >
   366→            {tab.icon} {tab.label}
   367→          </button>
   368→        ))}
   369→      </div>
   370→
   371→      {/* Error */}
   372→      {error && (
   373→        <div className="mb-4 p-4 bg-red-50 border border-red-200 rounded-lg text-red-700">
   374→          ❌ {error}
   375→        </div>
   376→      )}
   377→
   378→      {/* Content */}
   379→      <div className="space-y-6">
   380→        {activeTab === "simulate" && (
   381→          <SimulationTab
   382→            form={form}
   383→            setForm={setForm}
   384→            result={simResult}
   385→            chartData={simChartData}
   386→            loading={loading}
   387→            onRun={runSimulation}
   388→          />
   389→        )}
   390→        {activeTab === "protocol" && (
   391→          <ProtocolTab
   392→            form={form}
   393→            setForm={setForm}
   394→            result={protocolResult}
   395→            chartData={protocolChartData}
   396→            loading={loading}
   397→            onRun={runProtocol}
   398→          />
   399→        )}
   400→        {activeTab === "extrapolate" && (
   401→          <ExtrapolationTab
   402→            form={extrapForm}
   403→            setForm={setExtrapForm}
   404→            result={extrapResult}
   405→            loading={loading}
   406→            onRun={runExtrapolation}
   407→          />
   408→        )}
   409→        {activeTab === "monte-carlo" && (
   410→          <MonteCarloTab
   411→            form={mcForm}
   412→            setForm={setMcForm}
   413→            result={mcResult}
   414→            histogramData={mcHistogramData}
   415→            loading={loading}
   416→            onRun={runMonteCarlo}
   417→          />
   418→        )}
   419→        {activeTab === "molecular-risk" && (
   420→          <MolecularRiskTab
   421→            smiles={smilesInput}
   422→            setSmiles={setSmilesInput}
   423→            result={riskResult}
   424→            loading={loading}
   425→            onRun={runMolecularRisk}
   426→          />
   427→        )}
   428→      </div>
   429→    </div>
   430→  );
   431→};

   481→    {/* Chart + Results */}
   482→    <div className="lg:col-span-2 space-y-6">
