    90→  useEffect(() => {

   115→
   116→  // Apply source filter client-side on fetched molecules, then sort
   117→  const displayed = (() => {

   124→    else if (sortBy === 'molarMass') filtered.sort((a, b) => b.molarMass - a.molarMass)
   125→    else if (sortBy === 'risk') {
   126→      const riskOrder: Record<string, number> = { critical: 0, high: 1, moderate: 2, low: 3 }
   127→      filtered.sort((a, b) => (riskOrder[a.riskLevel] ?? 4) - (riskOrder[b.riskLevel] ?? 4))

   131→
   132→  // Count per risk level (from apiMolecules for filter pills)
   133→  const riskCounts = (() => {

   135→    apiMolecules.forEach((mol) => { counts[mol.riskLevel] = (counts[mol.riskLevel] || 0) + 1 })
   136→    return counts
   137→  })()
   138→

   145→      const res = await fetch(`/api/degradation-products?moleculeId=${mol.id}`)

   147→        const data = await res.json()

   150→    } catch { /* ignore */ }
   151→    setDegradationLoading(false)
   152→  }
   153→
   154→  const handleAddDegradation = async () => {

   174→        setNewDegradation({ name: '', smiles: '', percentage: 0, hazardLevel: 'low' })
   175→        // Refresh list

   178→          const data = await refreshRes.json()

   180→        }

   182→        const err = await res.json().catch(() => ({}))
   183→        toast({ title: 'Error', description: err.error || 'Failed to add degradation product', variant: 'destructive' })
   184→      }
   185→    } catch {
   186→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })

   191→
   192→  const handleImportCSV = async (e: React.ChangeEvent<HTMLInputElement>) => {
   193→    const file = e.target.files?.[0]
   194→    if (!file) return
   195→    setImporting(true)
   196→    try {
   197→      const text = await file.text()
   198→      const lines = text.split(/\r?\n/).filter((l) => l.trim())
   199→      if (lines.length < 2) {
   200→        toast({ title: 'Import failed', description: 'CSV must have a header row and at least one data row', variant: 'destructive' })
   201→        return
   202→      }
   203→      const parseCSVLine = (line: string): string[] => {
   204→        const out: string[] = []
   205→        let cur = ''
   206→        let inQ = false
   207→        for (let i = 0; i < line.length; i++) {
   208→          const ch = line[i]
   209→          if (ch === '"' && line[i + 1] === '"') { cur += '"'; i++ }
   210→          else if (ch === '"') { inQ = !inQ }
   211→          else if (ch === ',' && !inQ) { out.push(cur); cur = '' }
   212→          else { cur += ch }
   213→        }
   214→        out.push(cur)
   215→        return out
   216→      }
   217→      const headers = parseCSVLine(lines[0]).map((h) => h.trim().toLowerCase())
   218→      const nameIdx = headers.findIndex((h) => h === 'name')
   219→      if (nameIdx === -1) {
   220→        toast({ title: 'Import failed', description: 'CSV must have a "name" column', variant: 'destructive' })
   221→        return
   222→      }
   223→      const casIdx = headers.findIndex((h) => h === 'casnumber' || h === 'cas')
   224→      const formulaIdx = headers.findIndex((h) => h === 'formula')
   225→      const smilesIdx = headers.findIndex((h) => h === 'smiles')
   226→      const mwIdx = headers.findIndex((h) => h === 'molarmass' || h === 'mw')
   227→      const logPIdx = headers.findIndex((h) => h === 'logp')
   228→      const scoreIdx = headers.findIndex((h) => h === 'stabilityscore' || h === 'predictedstabilityscore' || h === 'score')
   229→      const riskIdx = headers.findIndex((h) => h === 'risklevel' || h === 'risk')
   230→
   231→      let success = 0, failed = 0
   232→      for (let i = 1; i < lines.length; i++) {
   233→        const cells = parseCSVLine(lines[i])
   234→        const name = cells[nameIdx]?.trim()
   235→        if (!name) { failed++; continue }
   236→        try {
   237→          const res = await fetch('/api/molecules', {
   238→            method: 'POST',
   239→            headers: { 'Content-Type': 'application/json' },
   240→            body: JSON.stringify({
   241→              name,
   242→              casNumber: casIdx >= 0 ? cells[casIdx]?.trim() || undefined : undefined,
   243→              formula: formulaIdx >= 0 ? cells[formulaIdx]?.trim() || undefined : undefined,
   244→              smiles: smilesIdx >= 0 ? cells[smilesIdx]?.trim() || undefined : undefined,
   245→              molarMass: mwIdx >= 0 ? parseFloat(cells[mwIdx]) || undefined : undefined,
   246→              logP: logPIdx >= 0 ? parseFloat(cells[logPIdx]) || undefined : undefined,
   247→              predictedStabilityScore: scoreIdx >= 0 ? parseFloat(cells[scoreIdx]) || undefined : undefined,
   248→              riskLevel: riskIdx >= 0 ? cells[riskIdx]?.trim().toLowerCase() || 'low' : 'low',
   249→            }),
   250→          })
   251→          if (res.ok) success++; else failed++
   252→        } catch { failed++ }
   253→      }
   254→      toast({
   255→        title: 'Import complete',
   256→        description: `Imported ${success} molecule${success !== 1 ? 's' : ''}${failed ? `, ${failed} failed` : ''}`,
   257→        variant: failed > 0 ? 'destructive' : 'default',
   258→      })
   259→      handleRefresh()
   260→    } catch {
   261→      toast({ title: 'Import failed', description: 'Failed to read file', variant: 'destructive' })
   262→    } finally {
   263→      setImporting(false)
   264→      if (fileInputRef.current) fileInputRef.current.value = ''
   265→    }
   266→  }
   267→
   268→  const handleRefresh = () => {

   272→

   284→    }

   288→        method: 'POST',

   291→          name: newMolecule.name,
   292→          casNumber: newMolecule.casNumber || undefined,
   293→          smiles: newMolecule.smiles || undefined,
   294→          formula: newMolecule.formula || undefined,
   295→          molarMass: newMolecule.molarMass || undefined,
   296→          logP: newMolecule.logP || undefined,
   297→          predictedStabilityScore: newMolecule.predictedStabilityScore || undefined,
   298→          riskLevel: newMolecule.riskLevel,
   299→          dataSource: newMolecule.dataSource,
   300→          description: newMolecule.description || undefined,

   302→      })

   306→        resetNewMolecule()
   307→        handleRefresh()
   308→      } else {
   309→        const errBody = await res.json().catch(() => ({}))
   310→        toast({ title: 'Error', description: errBody.error || 'Failed to create molecule', variant: 'destructive' })
   311→      }
   312→    } catch {
   313→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })

   315→      setCreating(false)
   316→    }

   326→          id: m.id,

   331→          molarMass: m.molarMass ?? '',
   332→          logP: m.logP ?? '',
   333→          stabilityScore: m.predictedStabilityScore ?? '',
   334→          riskLevel: m.riskLevel ?? '',
   335→          dataSource: m.dataSource ?? '',
   336→          meltingPoint: m.meltingPoint ?? '',
   337→          boilingPoint: m.boilingPoint ?? '',
   338→          description: m.description ?? '',
   339→        }))

   345→        }
   346→      } else {
   347→        toast({ title: 'Export failed', description: 'Failed to fetch molecules', variant: 'destructive' })
   348→      }
   349→    } catch {
   350→      toast({ title: 'Export failed', description: 'Network error', variant: 'destructive' })
   351→    } finally {
   352→      setExporting(false)
   353→    }

   361→      className="space-y-6"

   363→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">

   366→          <p className="text-muted-foreground">Browse and search chemical compounds with stability assessments</p>

   379→            type="file"

   384→          />

   420→            size="sm"
   421→            className={`h-7 text-xs rounded-full ${riskFilter === level ? (RISK_PILL_ACTIVE[level] || 'bg-emerald-600 text-white') : ''} ${RISK_PILL_OUTLINE[level] || ''}`}
   422→            onClick={() => { setRiskFilter(level); setCurrentPageNum(1) }}
   423→          >
   424→            {level === 'all' ? 'All' : level.charAt(0).toUpperCase() + level.slice(1)}
   425→            {level !== 'all' && <span className="ml-1 opacity-60">({riskCounts[level] || 0})</span>}

   437→            onChange={(e) => { setSearchQuery(e.target.value); setCurrentPageNum(1) }}
   438→            className="pl-9"
   439→          />

   444→          </SelectTrigger>
   445→          <SelectContent>
   446→            <SelectItem value="name">Name</SelectItem>
   447→            <SelectItem value="stability">Stability</SelectItem>
   448→            <SelectItem value="molarMass">Molar Mass</SelectItem>
   449→            <SelectItem value="risk">Risk Level</SelectItem>
   450→          </SelectContent>
   451→        </Select>
   452→        <Select value={riskFilter} onValueChange={(v) => { setRiskFilter(v); setCurrentPageNum(1) }}>
   453→          <SelectTrigger className="w-[140px]">
   454→            <SelectValue placeholder="Risk Level" />
   455→          </SelectTrigger>
   456→          <SelectContent>
   457→            <SelectItem value="all">All Risks</SelectItem>
   458→            <SelectItem value="low">Low</SelectItem>
   459→            <SelectItem value="moderate">Moderate</SelectItem>
   460→            <SelectItem value="high">High</SelectItem>
   461→            <SelectItem value="critical">Critical</SelectItem>
   462→          </SelectContent>
   463→        </Select>
   464→        <Select value={sourceFilter} onValueChange={(v) => { setSourceFilter(v); setCurrentPageNum(1) }}>
   465→          <SelectTrigger className="w-[140px]">
   466→            <SelectValue placeholder="Data Source" />
   467→          </SelectTrigger>
   468→          <SelectContent>
   469→            <SelectItem value="all">All Sources</SelectItem>
   470→            <SelectItem value="PubChem">PubChem</SelectItem>
   471→            <SelectItem value="ChEMBL">ChEMBL</SelectItem>
   472→            <SelectItem value="Manual">Manual</SelectItem>
   473→          </SelectContent>
   474→        </Select>
   475→      </div>
   476→
   477→      {/* Results count */}
   478→      <p className="text-sm text-muted-foreground">
   479→        {loading ? 'Loading...' : `Showing ${displayed.length} of ${totalCount} molecules`}
   480→      </p>
   481→
   482→      {/* Table / Grid View */}
   483→      {viewMode === 'table' ? (
   484→      <Card className="backdrop-blur-sm bg-card/80">
   485→        <CardContent className="p-0">
   486→          <div className="max-h-96 overflow-y-auto">
