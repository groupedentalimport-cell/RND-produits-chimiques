     1→'use client'
     2→
     3→import { useState, useEffect } from 'react'
     4→import { motion, AnimatePresence } from 'framer-motion'
     5→import {
     6→  Plus, RefreshCw, Download, ArrowRight, Thermometer, Droplets,
     7→  Clock, Beaker, Activity, Eye, Gauge, Shield, CheckCircle2,
     8→  XCircle, Trash2, List, CalendarRange, CheckSquare, X, GitCompareArrows,
     9→} from 'lucide-react'
    10→import {
    11→  Card, CardContent, CardFooter,
    12→} from '@/components/ui/card'
    13→import { Button } from '@/components/ui/button'
    14→import { Badge } from '@/components/ui/badge'
    15→import { Input } from '@/components/ui/input'
    16→import { Label } from '@/components/ui/label'
    17→import { Skeleton } from '@/components/ui/skeleton'
    18→import { Separator } from '@/components/ui/separator'
    19→import {
    20→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription, DialogFooter,
    21→} from '@/components/ui/dialog'
    22→import {
    23→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    24→} from '@/components/ui/select'
    25→import {
    26→  Table, TableHeader, TableBody, TableRow, TableHead, TableCell,
    27→} from '@/components/ui/table'
    28→import {
    29→  ToggleGroup, ToggleGroupItem,
    30→} from '@/components/ui/toggle-group'
    31→import { Checkbox } from '@/components/ui/checkbox'
    32→import { useToast } from '@/hooks/use-toast'
    33→import {
    34→  studyTypeLabels, statusColors, transformStudy, exportCSV,
    35→} from '@/lib/sample-data'
    36→import type { StudyData } from '@/lib/types'
    37→import { StudyTimeline } from './StudyTimeline'
    38→
    39→export function StudiesPage() {
    40→  const { toast } = useToast()
    41→  const [statusFilter, setStatusFilter] = useState('all')
    42→  const [typeFilter, setTypeFilter] = useState('all')
    43→  const [studyTypeFilter, setStudyTypeFilter] = useState('all')
    44→  const [view, setView] = useState<'list' | 'timeline'>('list')
    45→  const [createOpen, setCreateOpen] = useState(false)
    46→  const [loading, setLoading] = useState(true)
    47→  const [refreshKey, setRefreshKey] = useState(0)
    48→  const [apiStudies, setApiStudies] = useState<StudyData[]>([])
    49→  const [totalCount, setTotalCount] = useState(0)
    50→  const [studiesByStatus, setStudiesByStatus] = useState<{ status: string; _count: { status: number } }[]>([])
    51→  const [creating, setCreating] = useState(false)
    52→  const [exporting, setExporting] = useState(false)
    53→  const [detailOpen, setDetailOpen] = useState(false)
    54→  const [detailStudy, setDetailStudy] = useState<any | null>(null)
    55→  const [detailLoading, setDetailLoading] = useState(false)
    56→  const [signing, setSigning] = useState(false)
    57→  const [statusUpdating, setStatusUpdating] = useState(false)
    58→  const [newTimePoint, setNewTimePoint] = useState({ timeDays: 0, percentRemaining: 100, isOOS: false, isOOT: false })
    59→  const [addingTimePoint, setAddingTimePoint] = useState(false)
    60→  const [deletingTimePointId, setDeletingTimePointId] = useState<string | null>(null)
    61→  const [newStudy, setNewStudy] = useState({
    62→    substanceName: '',
    63→    studyType: 'long_term',
    64→    temperatureC: 25,
    65→    humidityPercent: 60,
    66→    durationMonths: 24,
    67→    ph: 7.0,
    68→  })
    69→  const [selectedIds, setSelectedIds] = useState<string[]>([])
    70→
    71→  const toggleSelect = (id: string) => {
    72→    setSelectedIds(prev => prev.includes(id) ? prev.filter(x => x !== id) : [...prev, id])
    73→  }
    74→  const clearSelection = () => setSelectedIds([])
    75→
    76→  // Batch status update handler
    77→  const batchUpdateStatus = async (newStatus: string) => {
    78→    if (selectedIds.length === 0) return
    79→    try {
    80→      const results = await Promise.all(selectedIds.map(id =>
    81→        fetch(`/api/studies/${id}`, {
    82→          method: 'PUT',
    83→          headers: { 'Content-Type': 'application/json' },
    84→          body: JSON.stringify({ status: newStatus }),
    85→        })
    86→      ))
    87→      const successCount = results.filter(r => r.ok).length
    88→      toast({
    89→        title: `Batch status update`,
    90→        description: `${successCount} of ${selectedIds.length} studies updated to "${newStatus.replace('_', ' ')}"`,
    91→      })
    92→      clearSelection()
    93→      handleRefresh()
    94→    } catch {
    95→      toast({ title: 'Error', description: 'Network error during batch update', variant: 'destructive' })
    96→    }
    97→  }
    98→
    99→  // Batch export selected studies
   100→  const batchExportCSV = async () => {
   101→    if (selectedIds.length === 0) return
   102→    try {
   103→      const res = await fetch('/api/studies?limit=1000')
   104→      if (res.ok) {
   105→        const data = await res.json()
   106→        const selectedStudies = (data.studies || []).filter((s: any) => selectedIds.includes(s.id))
   107→        const rows = selectedStudies.map((s: any) => ({
   108→          id: s.id,
   109→          studyCode: s.studyCode ?? '',
   110→          substanceName: s.substanceName ?? '',
   111→          studyType: s.studyType ?? '',
   112→          status: s.status ?? '',
   113→          temperatureC: s.temperatureC ?? '',
   114→          humidityPercent: s.humidityPercent ?? '',
   115→          durationMonths: s.durationMonths ?? '',
   116→          predictedShelfLifeMonths: s.predictedShelfLifeMonths ?? '',
   117→        }))
   118→        exportCSV(rows, `chemstab-selected-studies-${new Date().toISOString().slice(0, 10)}.csv`)
   119→        toast({ title: 'Export complete', description: `Exported ${rows.length} selected studies to CSV` })
   120→      }
   121→    } catch {
   122→      toast({ title: 'Export failed', description: 'Network error', variant: 'destructive' })
   123→    }
   124→  }
   125→
   126→  useEffect(() => {
   127→    let cancelled = false
   128→    const loadData = async () => {
   129→      const params = new URLSearchParams()
   130→      if (statusFilter !== 'all') params.set('status', statusFilter)
   131→      if (studyTypeFilter !== 'all') params.set('type', studyTypeFilter)
   132→      try {
   133→        const [studiesRes, statsRes] = await Promise.all([
   134→          fetch(`/api/studies?${params.toString()}`),
   135→          fetch('/api/stats'),
   136→        ])
   137→        if (studiesRes.ok && !cancelled) {
   138→          const data = await studiesRes.json()
   139→          const transformed: StudyData[] = (data.studies || []).map((s: any) => ({
   140→            ...transformStudy(s), humidityPercent: s.humidityPercent ?? null, ph: s.ph ?? null,
   141→          }))
   142→          if (!cancelled) {
   143→            setApiStudies(transformed)
   144→            setTotalCount(data.pagination?.total ?? 0)
   145→          }
   146→        }
   147→        if (statsRes.ok && !cancelled) {
   148→          const statsData = await statsRes.json()
   149→          if (statsData.studiesByStatus && !cancelled) {
   150→            setStudiesByStatus(statsData.studiesByStatus)
   151→          }
   152→        }
   153→      } catch { /* fallback */ }
   154→      if (!cancelled) setLoading(false)
   155→    }
   156→    loadData()
   157→    return () => { cancelled = true }
   158→  }, [statusFilter, studyTypeFilter, refreshKey])
   159→
   160→  const handleRefresh = () => setRefreshKey(k => k + 1)
   161→
   162→  const handleCreate = async () => {
   163→    if (!newStudy.substanceName.trim()) {
   164→      toast({ title: 'Validation error', description: 'Substance name is required', variant: 'destructive' })
   165→      return
   166→    }
   167→    setCreating(true)
   168→    // Generate a study code based on substance name + timestamp
   169→    const code = `STB-${new Date().getFullYear()}-${String(Date.now()).slice(-5)}`
   170→    try {
   171→      const res = await fetch('/api/studies', {
   172→        method: 'POST',
   173→        headers: { 'Content-Type': 'application/json' },
   174→        body: JSON.stringify({
   175→          studyCode: code,
   176→          substanceName: newStudy.substanceName,
   177→          studyType: newStudy.studyType,
   178→          temperatureC: newStudy.temperatureC,
   179→          humidityPercent: newStudy.humidityPercent,
   180→          durationMonths: newStudy.durationMonths,
   181→          ph: newStudy.ph,
   182→        }),
   183→      })
   184→      if (res.ok) {
   185→        toast({
   186→          title: 'Study created',
   187→          description: `${newStudy.substanceName} study created successfully (${code})`,
   188→        })
   189→        setCreateOpen(false)
   190→        setNewStudy({ substanceName: '', studyType: 'long_term', temperatureC: 25, humidityPercent: 60, durationMonths: 24, ph: 7.0 })
   191→        handleRefresh()
   192→      } else {
   193→        const errBody = await res.json().catch(() => ({}))
   194→        toast({ title: 'Error', description: errBody.error || 'Failed to create study', variant: 'destructive' })
   195→      }
   196→    } catch {
   197→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   198→    } finally {
   199→      setCreating(false)
   200→    }
   201→  }
   202→
   203→  const openDetail = async (study: StudyData) => {
   204→    setDetailOpen(true)
   205→    setDetailLoading(true)
   206→    setDetailStudy(null)
   207→    try {
   208→      const res = await fetch(`/api/studies/${study.id}`)
   209→      if (res.ok) {
   210→        const data = await res.json()
   211→        if (data.study) {
   212→          setDetailStudy(data.study)
   213→        } else {
   214→          // Fallback: use the row data we already have
   215→          setDetailStudy(study)
   216→        }
   217→      } else {
   218→        setDetailStudy(study)
   219→      }
   220→    } catch {
   221→      setDetailStudy(study)
   222→    } finally {
   223→      setDetailLoading(false)
   224→    }
   225→  }
   226→
   227→  const updateStudyStatus = async (newStatus: string) => {
   228→    if (!detailStudy) return
   229→    setStatusUpdating(true)
   230→    try {
   231→      const res = await fetch(`/api/studies/${detailStudy.id}`, {
   232→        method: 'PUT',
   233→        headers: { 'Content-Type': 'application/json' },
   234→        body: JSON.stringify({
   235→          status: newStatus,
   236→          rejectionReason: newStatus === 'rejected' ? 'Rejected during review' : undefined,
   237→        }),
   238→      })
   239→      if (res.ok) {
   240→        const data = await res.json()
   241→        if (data.study) {
   242→          setDetailStudy((prev: any) => prev ? { ...prev, ...data.study } : prev)
   243→        }
   244→        toast({ title: 'Status updated', description: `Study marked as ${newStatus.replace('_', ' ')}` })
   245→        handleRefresh()
   246→      } else {
   247→        toast({ title: 'Error', description: 'Failed to update status', variant: 'destructive' })
   248→      }
   249→    } catch {
   250→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   251→    } finally {
   252→      setStatusUpdating(false)
   253→    }
   254→  }
   255→
   256→  const signStudy = async () => {
   257→    if (!detailStudy) return
   258→    setSigning(true)
   259→    try {
   260→      const res = await fetch(`/api/studies/${detailStudy.id}`, {
   261→        method: 'POST',
   262→        headers: { 'Content-Type': 'application/json' },
   263→        body: JSON.stringify({
   264→          signerName: 'Dr. Sarah Chen',
   265→          signerRole: 'org_admin',
   266→          meaning: 'Reviewed and Approved',
   267→          newStatus: detailStudy.status === 'under_review' ? 'approved' : undefined,
   268→        }),
   269→      })
   270→      if (res.ok) {
   271→        const data = await res.json()
   272→        // Refresh detail to show new signature
   273→        if (data.study) {
   274→          setDetailStudy((prev: any) => prev ? { ...prev, ...data.study, signatures: [...(prev.signatures || []), data.signature] } : prev)
   275→        }
   276→        toast({ title: 'Study signed', description: 'Electronic signature recorded (FDA 21 CFR Part 11)' })
   277→        handleRefresh()
   278→      } else {
   279→        toast({ title: 'Error', description: 'Failed to sign study', variant: 'destructive' })
   280→      }
   281→    } catch {
   282→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   283→    } finally {
   284→      setSigning(false)
   285→    }
   286→  }
   287→
   288→  const handleAddTimePoint = async () => {
   289→    if (!detailStudy) return
   290→    if (newTimePoint.timeDays < 0) {
   291→      toast({ title: 'Validation error', description: 'Time (days) must be ≥ 0', variant: 'destructive' })
   292→      return
   293→    }
   294→    setAddingTimePoint(true)
   295→    try {
   296→      const degradationPercent = Math.max(0, 100 - newTimePoint.percentRemaining)
   297→      const res = await fetch('/api/timepoints', {
   298→        method: 'POST',
   299→        headers: { 'Content-Type': 'application/json' },
   300→        body: JSON.stringify({
   301→          timeDays: newTimePoint.timeDays,
   302→          percentRemaining: newTimePoint.percentRemaining,
   303→          degradationPercent,
   304→          isOOS: newTimePoint.isOOS,
   305→          isOOT: newTimePoint.isOOT,
   306→          studyId: detailStudy.id,
   307→        }),
   308→      })
   309→      if (res.ok) {
   310→        const data = await res.json()
   311→        toast({ title: 'Time point added', description: `Day ${newTimePoint.timeDays} recorded` })
   312→        setDetailStudy((prev: any) => prev
   313→          ? { ...prev, timePoints: [...(prev.timePoints || []), data.timePoint].sort((a: any, b: any) => a.timeDays - b.timeDays) }
   314→          : prev)
   315→        setNewTimePoint({ timeDays: 0, percentRemaining: 100, isOOS: false, isOOT: false })
   316→      } else {
   317→        const err = await res.json().catch(() => ({}))
   318→        toast({ title: 'Error', description: err.error || 'Failed to add time point', variant: 'destructive' })
   319→      }
   320→    } catch {
   321→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   322→    } finally {
   323→      setAddingTimePoint(false)
   324→    }
   325→  }
   326→
   327→  const handleDeleteTimePoint = async (tpId: string) => {
   328→    setDeletingTimePointId(tpId)
   329→    try {
   330→      const res = await fetch(`/api/timepoints/${tpId}`, { method: 'DELETE' })
   331→      if (res.ok) {
   332→        toast({ title: 'Time point deleted', description: 'Measurement removed from study' })
   333→        setDetailStudy((prev: any) => prev
   334→          ? { ...prev, timePoints: (prev.timePoints || []).filter((tp: any) => tp.id !== tpId) }
   335→          : prev)
   336→      } else {
   337→        toast({ title: 'Error', description: 'Failed to delete time point', variant: 'destructive' })
   338→      }
   339→    } catch {
   340→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   341→    } finally {
   342→      setDeletingTimePointId(null)
   343→    }
   344→  }
   345→
   346→  const exportStudiesCSV = async () => {
   347→    setExporting(true)
   348→    try {
   349→      const res = await fetch('/api/studies?limit=1000')
   350→      if (res.ok) {
   351→        const data = await res.json()
   352→        const rows = (data.studies || []).map((s: any) => ({
   353→          id: s.id,
   354→          studyCode: s.studyCode ?? '',
   355→          substanceName: s.substanceName ?? '',
   356→          studyType: s.studyType ?? '',
   357→          status: s.status ?? '',
   358→          temperatureC: s.temperatureC ?? '',
   359→          humidityPercent: s.humidityPercent ?? '',
   360→          durationMonths: s.durationMonths ?? '',
   361→          predictedShelfLifeMonths: s.predictedShelfLifeMonths ?? '',
   362→          ph: s.ph ?? '',
   363→        }))
   364→        if (!rows.length) {
   365→          toast({ title: 'Nothing to export', description: 'No studies found to export' })
   366→        } else {
   367→          exportCSV(rows, `chemstab-studies-${new Date().toISOString().slice(0, 10)}.csv`)
   368→          toast({ title: 'Export complete', description: `Exported ${rows.length} studies to CSV` })
   369→        }
   370→      } else {
   371→        toast({ title: 'Export failed', description: 'Failed to fetch studies', variant: 'destructive' })
   372→      }
   373→    } catch {
   374→      toast({ title: 'Export failed', description: 'Network error', variant: 'destructive' })
   375→    } finally {
   376→      setExporting(false)
   377→    }
   378→  }
   379→
   380→  return (
   381→    <motion.div
   382→      initial={{ opacity: 0, y: 20 }}
   383→      animate={{ opacity: 1, y: 0 }}
   384→      transition={{ duration: 0.4 }}
   385→      className="space-y-6"
   386→    >
   387→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   388→        <div>
   389→          <h1 className="text-2xl font-bold tracking-tight bg-gradient-to-r from-emerald-600 to-teal-600 dark:from-emerald-400 dark:to-teal-400 bg-clip-text text-transparent">Studies Management</h1>
   390→          <p className="text-muted-foreground">Manage and track stability studies across your organization</p>
   391→        </div>
   392→        <div className="flex gap-2 flex-wrap">
   393→          <Button variant="outline" size="sm" onClick={exportStudiesCSV} disabled={exporting}>
   394→            <Download className={`size-4 mr-1 ${exporting ? 'animate-pulse' : ''}`} /> Export CSV
   395→          </Button>
   396→          <Button variant="outline" size="sm" onClick={handleRefresh} disabled={loading}>
   397→            <RefreshCw className={`size-4 mr-1 ${loading ? 'animate-spin' : ''}`} /> Refresh
   398→          </Button>
   399→          <Button className="bg-emerald-600 hover:bg-emerald-700 text-white" onClick={() => setCreateOpen(true)}>
   400→            <Plus className="size-4 mr-2" /> Create Study
   401→          </Button>
   402→        </div>
   403→      </div>
   404→
   405→      {/* Status Pipeline Visualization */}
   406→      <div className="flex items-center gap-1 p-3 rounded-xl bg-gradient-to-r from-emerald-50/50 to-teal-50/50 dark:from-emerald-950/20 dark:to-teal-950/20 border">
   407→        {['draft', 'in_progress', 'under_review', 'approved'].map((stage, i) => {
   408→          const count = studiesByStatus?.find(s => s.status === stage)?._count?.status || 0
   409→          const stageColors: Record<string, string> = {
   410→            draft: 'bg-slate-400', in_progress: 'bg-teal-500', under_review: 'bg-amber-500', approved: 'bg-emerald-500'
   411→          }
   412→          const stageLabels: Record<string, string> = {
   413→            draft: 'Draft', in_progress: 'In Progress', under_review: 'Under Review', approved: 'Approved'
   414→          }
   415→          return (
   416→            <div key={stage} className="flex items-center gap-1">
   417→              {i > 0 && <ArrowRight className="size-3 text-muted-foreground" />}
   418→              <div className="flex items-center gap-1.5 px-2 py-1 rounded-full bg-white/60 dark:bg-card/60">
   419→                <span className={`size-2 rounded-full ${stageColors[stage]}`} />
   420→                <span className="text-xs font-medium">{stageLabels[stage]}</span>
   421→                <Badge variant="secondary" className="text-[10px] h-4 px-1">{count}</Badge>
   422→              </div>
   423→            </div>
   424→          )
   425→        })}
   426→      </div>
   427→
   428→      {/* Study Type Filter Pills */}
   429→      <div className="flex flex-wrap gap-2 mb-1">
   430→        {['all', 'long_term', 'accelerated', 'intermediate', 'stress'].map((type) => (
   431→          <Button
   432→            key={type}
   433→            variant={studyTypeFilter === type ? 'default' : 'outline'}
   434→            size="sm"
   435→            className={`h-7 text-xs rounded-full ${studyTypeFilter === type ? 'bg-emerald-600 text-white hover:bg-emerald-700' : ''}`}
   436→            onClick={() => setStudyTypeFilter(type)}
   437→          >
   438→            {type === 'all' ? 'All Types' : studyTypeLabels[type]}
   439→          </Button>
   440→        ))}
   441→      </div>
   442→
   443→      {/* Status Filter + View Toggle */}
   444→      <div className="flex flex-wrap items-center justify-between gap-3">
   445→        <Select value={statusFilter} onValueChange={setStatusFilter}>
   446→          <SelectTrigger className="w-[160px]">
   447→            <SelectValue placeholder="Filter by Status" />
   448→          </SelectTrigger>
   449→          <SelectContent>
   450→            <SelectItem value="all">All Statuses</SelectItem>
   451→            <SelectItem value="draft">Draft</SelectItem>
   452→            <SelectItem value="in_progress">In Progress</SelectItem>
   453→            <SelectItem value="completed">Completed</SelectItem>
   454→            <SelectItem value="under_review">Under Review</SelectItem>
   455→            <SelectItem value="approved">Approved</SelectItem>
   456→          </SelectContent>
   457→        </Select>
   458→        <ToggleGroup
   459→          type="single"
   460→          value={view}
   461→          onValueChange={(v) => { if (v === 'list' || v === 'timeline') setView(v) }}
   462→          variant="outline"
   463→          size="sm"
   464→          className="rounded-lg"
   465→          aria-label="Studies view toggle"
   466→        >
   467→          <ToggleGroupItem value="list" className="gap-1.5 px-3 data-[state=on]:bg-emerald-600 data-[state=on]:text-white data-[state=on]:hover:bg-emerald-700">
   468→            <List className="size-3.5" /> List View
   469→          </ToggleGroupItem>
   470→          <ToggleGroupItem value="timeline" className="gap-1.5 px-3 data-[state=on]:bg-emerald-600 data-[state=on]:text-white data-[state=on]:hover:bg-emerald-700">
   471→            <CalendarRange className="size-3.5" /> Timeline View
   472→          </ToggleGroupItem>
   473→        </ToggleGroup>
   474→      </div>
   475→
   476→      {/* Studies Table (List View) */}
   477→      {view === 'list' && (
   478→      <Card className="backdrop-blur-sm bg-card/80">
   479→        <CardContent className="p-0">
   480→          <div className="max-h-96 overflow-y-auto">
   481→            {loading ? (
   482→              <div className="space-y-2 p-4">
   483→                {Array.from({ length: 5 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   484→              </div>
   485→            ) : (
   486→              <Table>
   487→                <TableHeader>
   488→                  <TableRow>
   489→                    <TableHead className="w-[40px]">Select</TableHead>
   490→                    <TableHead>Study Code</TableHead>
   491→                    <TableHead>Substance</TableHead>
   492→                    <TableHead>Type</TableHead>
   493→                    <TableHead>Temp (°C)</TableHead>
   494→                    <TableHead>Shelf Life</TableHead>
   495→                    <TableHead>Status</TableHead>
   496→                  </TableRow>
   497→                </TableHeader>
   498→                <TableBody>
   499→                  {apiStudies.map((std, idx) => {
   500→                    const isSelected = selectedIds.includes(std.id)
   501→                    return (
   502→                    <TableRow
   503→                      key={std.id}
   504→                      className={`cursor-pointer transition-all ${isSelected ? 'bg-emerald-50/60 dark:bg-emerald-900/15 shadow-[inset_3px_0_0_0_rgb(16,185,129)]' : idx % 2 === 1 ? 'bg-muted/30' : ''} hover:bg-emerald-50/50 dark:hover:bg-emerald-900/10`}
