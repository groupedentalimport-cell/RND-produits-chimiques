    43→
    44→export function MoleculesPage() {
    45→  const { toast } = useToast()
    46→  const { setPage } = useAppStore()
    47→  const [searchQuery, setSearchQuery] = useState('')
    48→  const [riskFilter, setRiskFilter] = useState('all')
    49→  const [sourceFilter, setSourceFilter] = useState('all')
    50→  const [currentPageNum, setCurrentPageNum] = useState(1)
    51→  const [selectedMolecule, setSelectedMolecule] = useState<MoleculeData | null>(null)
    52→  const [detailOpen, setDetailOpen] = useState(false)
    53→  const [loading, setLoading] = useState(true)
    54→  const [refreshKey, setRefreshKey] = useState(0)
    55→  const [apiMolecules, setApiMolecules] = useState<MoleculeData[]>([])
    56→  const [totalPages, setTotalPages] = useState(1)
    57→  const [totalCount, setTotalCount] = useState(0)
    58→  const [addOpen, setAddOpen] = useState(false)
    59→  const [creating, setCreating] = useState(false)
    60→  const [exporting, setExporting] = useState(false)
    61→  const [degradationProducts, setDegradationProducts] = useState<any[]>([])
    62→  const [degradationLoading, setDegradationLoading] = useState(false)
    63→  const [newDegradation, setNewDegradation] = useState({ name: '', smiles: '', percentage: 0, hazardLevel: 'low' })
    64→  const [addingDegradation, setAddingDegradation] = useState(false)
    65→  const [importing, setImporting] = useState(false)
    66→  const [sortBy, setSortBy] = useState('name')
    67→  const [viewMode, setViewMode] = useState<'table' | 'grid'>('table')
    68→  const fileInputRef = useRef<HTMLInputElement | null>(null)
    69→  const [newMolecule, setNewMolecule] = useState({
    70→    name: '',
    71→    casNumber: '',
    72→    smiles: '',
    73→    formula: '',
    74→    molarMass: 0,
    75→    logP: 0,
    76→    predictedStabilityScore: 0,
    77→    riskLevel: 'low',
    78→    dataSource: 'manual',
    79→    description: '',
    80→  })
    81→  const pageSize = 5
    82→
    83→  useEffect(() => {
    84→    let cancelled = false
    85→    const loadData = async () => {
    86→      const params = new URLSearchParams()
    87→      if (searchQuery) params.set('q', searchQuery)
    88→      if (riskFilter !== 'all') params.set('risk', riskFilter)
    89→      params.set('page', String(currentPageNum))
    90→      params.set('limit', String(pageSize))
    91→      try {
    92→        const res = await fetch(`/api/molecules?${params.toString()}`)
    93→        if (res.ok && !cancelled) {
    94→          const data = await res.json()
    95→          const transformed: MoleculeData[] = (data.molecules || []).map(transformMolecule)
    96→          if (!cancelled) {
    97→            setApiMolecules(transformed)
    98→            setTotalPages(data.pagination?.totalPages ?? 1)
    99→            setTotalCount(data.pagination?.total ?? 0)
   100→          }
   101→        }
   102→      } catch { /* fallback */ }
   103→      if (!cancelled) setLoading(false)
   104→    }
   105→    loadData()
   106→    return () => { cancelled = true }
   107→  }, [searchQuery, riskFilter, currentPageNum, refreshKey])
   108→
   109→  // Apply source filter client-side on fetched molecules, then sort
   110→  const displayed = (() => {
   111→    let filtered = sourceFilter === 'all'
   112→      ? apiMolecules
   113→      : apiMolecules.filter((mol) => mol.dataSource.toLowerCase() === sourceFilter.toLowerCase())
   114→    // Sort
   115→    if (sortBy === 'name') filtered.sort((a, b) => a.name.localeCompare(b.name))
   116→    else if (sortBy === 'stability') filtered.sort((a, b) => b.stabilityScore - a.stabilityScore)
   117→    else if (sortBy === 'molarMass') filtered.sort((a, b) => b.molarMass - a.molarMass)
   118→    else if (sortBy === 'risk') {
   119→      const riskOrder: Record<string, number> = { critical: 0, high: 1, moderate: 2, low: 3 }
   120→      filtered.sort((a, b) => (riskOrder[a.riskLevel] ?? 4) - (riskOrder[b.riskLevel] ?? 4))
   121→    }
   122→    return filtered
   123→  })()
   124→
   125→  // Count per risk level (from apiMolecules for filter pills)
   126→  const riskCounts = (() => {
   127→    const counts: Record<string, number> = { low: 0, moderate: 0, high: 0, critical: 0 }
   128→    apiMolecules.forEach((mol) => { counts[mol.riskLevel] = (counts[mol.riskLevel] || 0) + 1 })
   129→    return counts
   130→  })()
   131→
   132→  const openDetail = async (mol: MoleculeData) => {
   133→    setSelectedMolecule(mol)
   134→    setDetailOpen(true)
   135→    setDegradationProducts([])
   136→    setDegradationLoading(true)
   137→    try {
   138→      const res = await fetch(`/api/degradation-products?moleculeId=${mol.id}`)
   139→      if (res.ok) {
   140→        const data = await res.json()
   141→        setDegradationProducts(data.products || [])
   142→      }
   143→    } catch { /* ignore */ }
   144→    setDegradationLoading(false)
   145→  }
   146→
   147→  const handleAddDegradation = async () => {
   148→    if (!selectedMolecule || !newDegradation.name.trim()) {
   149→      toast({ title: 'Validation error', description: 'Product name is required', variant: 'destructive' })
   150→      return
   151→    }
   152→    setAddingDegradation(true)
   153→    try {
   154→      const res = await fetch('/api/degradation-products', {
   155→        method: 'POST',
   156→        headers: { 'Content-Type': 'application/json' },
   157→        body: JSON.stringify({
   158→          name: newDegradation.name,
   159→          smiles: newDegradation.smiles || undefined,
   160→          percentage: newDegradation.percentage || undefined,
   161→          hazardLevel: newDegradation.hazardLevel,
   162→          moleculeId: selectedMolecule.id,
   163→        }),
   164→      })
   165→      if (res.ok) {
   166→        toast({ title: 'Degradation product added', description: `${newDegradation.name} linked to ${selectedMolecule.name}` })
   167→        setNewDegradation({ name: '', smiles: '', percentage: 0, hazardLevel: 'low' })
   168→        // Refresh list
   169→        const refreshRes = await fetch(`/api/degradation-products?moleculeId=${selectedMolecule.id}`)
   170→        if (refreshRes.ok) {
   171→          const data = await refreshRes.json()
   172→          setDegradationProducts(data.products || [])
   173→        }
   174→      } else {
   175→        const err = await res.json().catch(() => ({}))
   176→        toast({ title: 'Error', description: err.error || 'Failed to add degradation product', variant: 'destructive' })
   177→      }
   178→    } catch {
   179→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   180→    } finally {
   181→      setAddingDegradation(false)
   182→    }
   183→  }
   184→
   185→  const handleImportCSV = async (e: React.ChangeEvent<HTMLInputElement>) => {
   186→    const file = e.target.files?.[0]
   187→    if (!file) return
   188→    setImporting(true)
   189→    try {
   190→      const text = await file.text()
   191→      const lines = text.split(/\r?\n/).filter((l) => l.trim())
   192→      if (lines.length < 2) {
   193→        toast({ title: 'Import failed', description: 'CSV must have a header row and at least one data row', variant: 'destructive' })
   194→        return
   195→      }
   196→      const parseCSVLine = (line: string): string[] => {
   197→        const out: string[] = []
   198→        let cur = ''
   199→        let inQ = false
   200→        for (let i = 0; i < line.length; i++) {
   201→          const ch = line[i]
   202→          if (ch === '"' && line[i + 1] === '"') { cur += '"'; i++ }
   203→          else if (ch === '"') { inQ = !inQ }
   204→          else if (ch === ',' && !inQ) { out.push(cur); cur = '' }
   205→          else { cur += ch }
   206→        }
   207→        out.push(cur)
   208→        return out
   209→      }
   210→      const headers = parseCSVLine(lines[0]).map((h) => h.trim().toLowerCase())
   211→      const nameIdx = headers.findIndex((h) => h === 'name')
   212→      if (nameIdx === -1) {
   213→        toast({ title: 'Import failed', description: 'CSV must have a "name" column', variant: 'destructive' })
   214→        return
   215→      }
   216→      const casIdx = headers.findIndex((h) => h === 'casnumber' || h === 'cas')
   217→      const formulaIdx = headers.findIndex((h) => h === 'formula')
   218→      const smilesIdx = headers.findIndex((h) => h === 'smiles')
   219→      const mwIdx = headers.findIndex((h) => h === 'molarmass' || h === 'mw')
   220→      const logPIdx = headers.findIndex((h) => h === 'logp')
   221→      const scoreIdx = headers.findIndex((h) => h === 'stabilityscore' || h === 'predictedstabilityscore' || h === 'score')
   222→      const riskIdx = headers.findIndex((h) => h === 'risklevel' || h === 'risk')
   223→
   224→      let success = 0, failed = 0
   225→      for (let i = 1; i < lines.length; i++) {
   226→        const cells = parseCSVLine(lines[i])
   227→        const name = cells[nameIdx]?.trim()
   228→        if (!name) { failed++; continue }
   229→        try {
   230→          const res = await fetch('/api/molecules', {
   231→            method: 'POST',
   232→            headers: { 'Content-Type': 'application/json' },
   233→            body: JSON.stringify({
   234→              name,
   235→              casNumber: casIdx >= 0 ? cells[casIdx]?.trim() || undefined : undefined,
   236→              formula: formulaIdx >= 0 ? cells[formulaIdx]?.trim() || undefined : undefined,
   237→              smiles: smilesIdx >= 0 ? cells[smilesIdx]?.trim() || undefined : undefined,
   238→              molarMass: mwIdx >= 0 ? parseFloat(cells[mwIdx]) || undefined : undefined,
   239→              logP: logPIdx >= 0 ? parseFloat(cells[logPIdx]) || undefined : undefined,
   240→              predictedStabilityScore: scoreIdx >= 0 ? parseFloat(cells[scoreIdx]) || undefined : undefined,
   241→              riskLevel: riskIdx >= 0 ? cells[riskIdx]?.trim().toLowerCase() || 'low' : 'low',
   242→            }),
   243→          })
   244→          if (res.ok) success++; else failed++
   245→        } catch { failed++ }
   246→      }
   247→      toast({
   248→        title: 'Import complete',
   249→        description: `Imported ${success} molecule${success !== 1 ? 's' : ''}${failed ? `, ${failed} failed` : ''}`,
   250→        variant: failed > 0 ? 'destructive' : 'default',
   251→      })
   252→      handleRefresh()
   253→    } catch {
   254→      toast({ title: 'Import failed', description: 'Failed to read file', variant: 'destructive' })
   255→    } finally {
   256→      setImporting(false)
   257→      if (fileInputRef.current) fileInputRef.current.value = ''
   258→    }
   259→  }
   260→
   261→  const handleRefresh = () => {
   262→    setLoading(true)
   263→    setRefreshKey(k => k + 1)
   264→  }
   265→
   266→  const resetNewMolecule = () => {
   267→    setNewMolecule({
   268→      name: '', casNumber: '', smiles: '', formula: '', molarMass: 0, logP: 0,
   269→      predictedStabilityScore: 0, riskLevel: 'low', dataSource: 'manual', description: '',
   270→    })
   271→  }
   272→
   273→  const handleAddMolecule = async () => {
   274→    if (!newMolecule.name.trim()) {
   275→      toast({ title: 'Validation error', description: 'Molecule name is required', variant: 'destructive' })
   276→      return
   277→    }
   278→    setCreating(true)
   279→    try {
   280→      const res = await fetch('/api/molecules', {
   281→        method: 'POST',
   282→        headers: { 'Content-Type': 'application/json' },
   283→        body: JSON.stringify({
   284→          name: newMolecule.name,
   285→          casNumber: newMolecule.casNumber || undefined,
   286→          smiles: newMolecule.smiles || undefined,
   287→          formula: newMolecule.formula || undefined,
   288→          molarMass: newMolecule.molarMass || undefined,
   289→          logP: newMolecule.logP || undefined,
   290→          predictedStabilityScore: newMolecule.predictedStabilityScore || undefined,
   291→          riskLevel: newMolecule.riskLevel,
   292→          dataSource: newMolecule.dataSource,
   293→          description: newMolecule.description || undefined,
   294→        }),
   295→      })
   296→      if (res.ok) {
   297→        toast({ title: 'Molecule added', description: `${newMolecule.name} created successfully` })
   298→        setAddOpen(false)
   299→        resetNewMolecule()
   300→        handleRefresh()
   301→      } else {
   302→        const errBody = await res.json().catch(() => ({}))
   303→        toast({ title: 'Error', description: errBody.error || 'Failed to create molecule', variant: 'destructive' })
   304→      }
   305→    } catch {
   306→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   307→    } finally {
   308→      setCreating(false)
   309→    }
   310→  }
   311→
   312→  const exportMoleculesCSV = async () => {
   313→    setExporting(true)
   314→    try {
   315→      const res = await fetch('/api/molecules?limit=1000')
   316→      if (res.ok) {
   317→        const data = await res.json()
   318→        const rows = (data.molecules || []).map((m: any) => ({
   319→          id: m.id,
   320→          name: m.name,
   321→          casNumber: m.casNumber ?? '',
   322→          smiles: m.smiles ?? '',
   323→          formula: m.formula ?? '',
   324→          molarMass: m.molarMass ?? '',
   325→          logP: m.logP ?? '',
   326→          stabilityScore: m.predictedStabilityScore ?? '',
   327→          riskLevel: m.riskLevel ?? '',
   328→          dataSource: m.dataSource ?? '',
   329→          meltingPoint: m.meltingPoint ?? '',
   330→          boilingPoint: m.boilingPoint ?? '',
   331→          description: m.description ?? '',
   332→        }))
   333→        if (!rows.length) {
   334→          toast({ title: 'Nothing to export', description: 'No molecules found to export' })
   335→        } else {
   336→          exportCSV(rows, `chemstab-molecules-${new Date().toISOString().slice(0, 10)}.csv`)
   337→          toast({ title: 'Export complete', description: `Exported ${rows.length} molecules to CSV` })
   338→        }
   339→      } else {
   340→        toast({ title: 'Export failed', description: 'Failed to fetch molecules', variant: 'destructive' })
   341→      }
   342→    } catch {
   343→      toast({ title: 'Export failed', description: 'Network error', variant: 'destructive' })
   344→    } finally {
   345→      setExporting(false)
   346→    }
   347→  }
   348→
   349→  return (
   350→    <motion.div
   351→      initial={{ opacity: 0, y: 20 }}
   352→      animate={{ opacity: 1, y: 0 }}
   353→      transition={{ duration: 0.4 }}
   354→      className="space-y-6"
   355→    >
   356→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   357→        <div>
   358→          <h1 className="text-2xl font-bold tracking-tight bg-gradient-to-r from-emerald-600 to-teal-600 dark:from-emerald-400 dark:to-teal-400 bg-clip-text text-transparent">Molecules Database</h1>
   359→          <p className="text-muted-foreground">Browse and search chemical compounds with stability assessments</p>
   360→        </div>
   361→        <div className="flex gap-2 flex-wrap items-center">
   362→          {/* View Mode Toggle */}
   363→          <div className="flex gap-1 mr-1">
   364→            <Button variant={viewMode === 'table' ? 'default' : 'outline'} size="sm" onClick={() => setViewMode('table')} className={viewMode === 'table' ? 'bg-emerald-600 text-white' : ''}>
   365→              <BarChart3 className="size-4" /> Table
   366→            </Button>
   367→            <Button variant={viewMode === 'grid' ? 'default' : 'outline'} size="sm" onClick={() => setViewMode('grid')} className={viewMode === 'grid' ? 'bg-emerald-600 text-white' : ''}>
   368→              <LayoutGrid className="size-4" /> Grid
   369→            </Button>
   370→          </div>
   371→          <input
   372→            type="file"
   373→            accept=".csv"
   374→            ref={fileInputRef}
   375→            onChange={handleImportCSV}
   376→            className="hidden"
   377→          />
   378→          <TooltipProvider delayDuration={200}>
   379→            <UiTooltip>
   380→              <TooltipTrigger asChild>
   381→                <Button variant="outline" size="sm" onClick={() => fileInputRef.current?.click()} disabled={importing}>
   382→                  <Download className={`size-4 mr-1 rotate-180 ${importing ? 'animate-pulse' : ''}`} /> Import CSV
   383→                </Button>
   384→              </TooltipTrigger>
   385→              <TooltipContent>Bulk import molecules from a CSV file (name column required)</TooltipContent>
   386→            </UiTooltip>
   387→          </TooltipProvider>
   388→          <TooltipProvider delayDuration={200}>
   389→            <UiTooltip>
   390→              <TooltipTrigger asChild>
   391→                <Button variant="outline" size="sm" onClick={exportMoleculesCSV} disabled={exporting}>
   392→                  <Download className={`size-4 mr-1 ${exporting ? 'animate-pulse' : ''}`} /> Export CSV
   393→                </Button>
   394→              </TooltipTrigger>
   395→              <TooltipContent>Download all molecules as a CSV file</TooltipContent>
   396→            </UiTooltip>
   397→          </TooltipProvider>
   398→          <Button variant="outline" size="sm" onClick={handleRefresh} disabled={loading}>
   399→            <RefreshCw className={`size-4 mr-1 ${loading ? 'animate-spin' : ''}`} /> Refresh
   400→          </Button>
   401→          <Button className="bg-emerald-600 hover:bg-emerald-700 text-white" onClick={() => setAddOpen(true)}>
   402→            <Plus className="size-4 mr-2" /> Add Molecule
   403→          </Button>
   404→        </div>
   405→      </div>
   406→
   407→      {/* Quick Filter Pills */}
   408→      <div className="flex flex-wrap gap-2">
   409→        {['all', 'low', 'moderate', 'high', 'critical'].map((level) => (
   410→          <Button
   411→            key={level}
   412→            variant={riskFilter === level ? 'default' : 'outline'}
   413→            size="sm"
   414→            className={`h-7 text-xs rounded-full ${riskFilter === level ? (RISK_PILL_ACTIVE[level] || 'bg-emerald-600 text-white') : ''} ${RISK_PILL_OUTLINE[level] || ''}`}
   415→            onClick={() => { setRiskFilter(level); setCurrentPageNum(1) }}
   416→          >
   417→            {level === 'all' ? 'All' : level.charAt(0).toUpperCase() + level.slice(1)}
   418→            {level !== 'all' && <span className="ml-1 opacity-60">({riskCounts[level] || 0})</span>}
   419→          </Button>
   420→        ))}
   421→      </div>
   422→
   423→      {/* Search & Filters */}
   424→      <div className="flex flex-col sm:flex-row gap-3">
   425→        <div className="relative flex-1">
   426→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   427→          <Input
   428→            placeholder="Search by name, CAS, SMILES..."
   429→            value={searchQuery}
   430→            onChange={(e) => { setSearchQuery(e.target.value); setCurrentPageNum(1) }}
   431→            className="pl-9"
   432→          />
   433→        </div>
   434→        <Select value={sortBy} onValueChange={setSortBy}>
   435→          <SelectTrigger className="w-[130px]">
   436→            <SelectValue placeholder="Sort by" />
   437→          </SelectTrigger>
   438→          <SelectContent>
   439→            <SelectItem value="name">Name</SelectItem>
   440→            <SelectItem value="stability">Stability</SelectItem>
   441→            <SelectItem value="molarMass">Molar Mass</SelectItem>
   442→            <SelectItem value="risk">Risk Level</SelectItem>
   443→          </SelectContent>
   444→        </Select>
   445→        <Select value={riskFilter} onValueChange={(v) => { setRiskFilter(v); setCurrentPageNum(1) }}>
   446→          <SelectTrigger className="w-[140px]">
   447→            <SelectValue placeholder="Risk Level" />
   448→          </SelectTrigger>
   449→          <SelectContent>
   450→            <SelectItem value="all">All Risks</SelectItem>
   451→            <SelectItem value="low">Low</SelectItem>
   452→            <SelectItem value="moderate">Moderate</SelectItem>
   453→            <SelectItem value="high">High</SelectItem>
   454→            <SelectItem value="critical">Critical</SelectItem>
   455→          </SelectContent>
   456→        </Select>
   457→        <Select value={sourceFilter} onValueChange={(v) => { setSourceFilter(v); setCurrentPageNum(1) }}>
   458→          <SelectTrigger className="w-[140px]">
   459→            <SelectValue placeholder="Data Source" />
   460→          </SelectTrigger>
   461→          <SelectContent>
   462→            <SelectItem value="all">All Sources</SelectItem>
   463→            <SelectItem value="PubChem">PubChem</SelectItem>
   464→            <SelectItem value="ChEMBL">ChEMBL</SelectItem>
   465→            <SelectItem value="Manual">Manual</SelectItem>
   466→          </SelectContent>
   467→        </Select>
   468→      </div>
   469→
   470→      {/* Results count */}
   471→      <p className="text-sm text-muted-foreground">
   472→        {loading ? 'Loading...' : `Showing ${displayed.length} of ${totalCount} molecules`}
   473→      </p>
   474→
   475→      {/* Table / Grid View */}
   476→      {viewMode === 'table' ? (
   477→      <Card className="backdrop-blur-sm bg-card/80">
   478→        <CardContent className="p-0">
   479→          <div className="max-h-96 overflow-y-auto">
   480→            {loading ? (
   481→              <div className="space-y-2 p-4">
   482→                {Array.from({ length: 5 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   483→              </div>
   484→            ) : (
   485→              <Table>
   486→                <TableHeader>
   487→                  <TableRow>
   488→                    <TableHead>Name</TableHead>
   489→                    <TableHead>Formula</TableHead>
   490→                    <TableHead>MW</TableHead>
