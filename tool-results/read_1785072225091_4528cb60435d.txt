     1→     1→'use client'
     2→     2→
     3→     3→import { useState, useEffect } from 'react'
     4→     4→import { motion } from 'framer-motion'
     5→     5→import {
     6→     6→  Plus, RefreshCw, Download, ArrowRight, Thermometer, Droplets,
     7→     7→  Clock, Beaker, Activity, Eye, Gauge, Shield, CheckCircle2,
     8→     8→  XCircle, Trash2,
     9→     9→} from 'lucide-react'
    10→    10→import {
    11→    11→  Card, CardContent, CardFooter,
    12→    12→} from '@/components/ui/card'
    13→    13→import { Button } from '@/components/ui/button'
    14→    14→import { Badge } from '@/components/ui/badge'
    15→    15→import { Input } from '@/components/ui/input'
    16→    16→import { Label } from '@/components/ui/label'
    17→    17→import { Skeleton } from '@/components/ui/skeleton'
    18→    18→import { Separator } from '@/components/ui/separator'
    19→    19→import {
    20→    20→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription, DialogFooter,
    21→    21→} from '@/components/ui/dialog'
    22→    22→import {
    23→    23→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    24→    24→} from '@/components/ui/select'
    25→    25→import {
    26→    26→  Table, TableHeader, TableBody, TableRow, TableHead, TableCell,
    27→    27→} from '@/components/ui/table'
    28→    28→import { useToast } from '@/hooks/use-toast'
    29→    29→import {
    30→    30→  studyTypeLabels, statusColors, transformStudy, exportCSV,
    31→    31→} from '@/lib/sample-data'
    32→    32→import type { StudyData } from '@/lib/types'
    33→    33→
    34→    34→export function StudiesPage() {
    35→    35→  const { toast } = useToast()
    36→    36→  const [statusFilter, setStatusFilter] = useState('all')
    37→    37→  const [typeFilter, setTypeFilter] = useState('all')
    38→    38→  const [studyTypeFilter, setStudyTypeFilter] = useState('all')
    39→    39→  const [createOpen, setCreateOpen] = useState(false)
    40→    40→  const [loading, setLoading] = useState(true)
    41→    41→  const [refreshKey, setRefreshKey] = useState(0)
    42→    42→  const [apiStudies, setApiStudies] = useState<StudyData[]>([])
    43→    43→  const [totalCount, setTotalCount] = useState(0)
    44→    44→  const [studiesByStatus, setStudiesByStatus] = useState<{ status: string; _count: { status: number } }[]>([])
    45→    45→  const [creating, setCreating] = useState(false)
    46→    46→  const [exporting, setExporting] = useState(false)
    47→    47→  const [detailOpen, setDetailOpen] = useState(false)
    48→    48→  const [detailStudy, setDetailStudy] = useState<any | null>(null)
    49→    49→  const [detailLoading, setDetailLoading] = useState(false)
    50→    50→  const [signing, setSigning] = useState(false)
    51→    51→  const [statusUpdating, setStatusUpdating] = useState(false)
    52→    52→  const [newTimePoint, setNewTimePoint] = useState({ timeDays: 0, percentRemaining: 100, isOOS: false, isOOT: false })
    53→    53→  const [addingTimePoint, setAddingTimePoint] = useState(false)
    54→    54→  const [deletingTimePointId, setDeletingTimePointId] = useState<string | null>(null)
    55→    55→  const [newStudy, setNewStudy] = useState({
    56→    56→    substanceName: '',
    57→    57→    studyType: 'long_term',
    58→    58→    temperatureC: 25,
    59→    59→    humidityPercent: 60,
    60→    60→    durationMonths: 24,
    61→    61→    ph: 7.0,
    62→    62→  })
    63→    63→
    64→    64→  useEffect(() => {
    65→    65→    let cancelled = false
    66→    66→    const loadData = async () => {
    67→    67→      const params = new URLSearchParams()
    68→    68→      if (statusFilter !== 'all') params.set('status', statusFilter)
    69→    69→      if (studyTypeFilter !== 'all') params.set('type', studyTypeFilter)
    70→    70→      try {
    71→    71→        const [studiesRes, statsRes] = await Promise.all([
    72→    72→          fetch(`/api/studies?${params.toString()}`),
    73→    73→          fetch('/api/stats'),
    74→    74→        ])
    75→    75→        if (studiesRes.ok && !cancelled) {
    76→    76→          const data = await studiesRes.json()
    77→    77→          const transformed: StudyData[] = (data.studies || []).map((s: any) => ({
    78→    78→            ...transformStudy(s), humidityPercent: s.humidityPercent ?? null, ph: s.ph ?? null,
    79→    79→          }))
    80→    80→          if (!cancelled) {
    81→    81→            setApiStudies(transformed)
    82→    82→            setTotalCount(data.pagination?.total ?? 0)
    83→    83→          }
    84→    84→        }
    85→    85→        if (statsRes.ok && !cancelled) {
    86→    86→          const statsData = await statsRes.json()
    87→    87→          if (statsData.studiesByStatus && !cancelled) {
    88→    88→            setStudiesByStatus(statsData.studiesByStatus)
    89→    89→          }
    90→    90→        }
    91→    91→      } catch { /* fallback */ }
    92→    92→      if (!cancelled) setLoading(false)
    93→    93→    }
    94→    94→    loadData()
    95→    95→    return () => { cancelled = true }
    96→    96→  }, [statusFilter, studyTypeFilter, refreshKey])
    97→    97→
    98→    98→  const handleRefresh = () => setRefreshKey(k => k + 1)
    99→    99→
   100→   100→  const handleCreate = async () => {
   101→   101→    if (!newStudy.substanceName.trim()) {
   102→   102→      toast({ title: 'Validation error', description: 'Substance name is required', variant: 'destructive' })
   103→   103→      return
   104→   104→    }
   105→   105→    setCreating(true)
   106→   106→    // Generate a study code based on substance name + timestamp
   107→   107→    const code = `STB-${new Date().getFullYear()}-${String(Date.now()).slice(-5)}`
   108→   108→    try {
   109→   109→      const res = await fetch('/api/studies', {
   110→   110→        method: 'POST',
   111→   111→        headers: { 'Content-Type': 'application/json' },
   112→   112→        body: JSON.stringify({
   113→   113→          studyCode: code,
   114→   114→          substanceName: newStudy.substanceName,
   115→   115→          studyType: newStudy.studyType,
   116→   116→          temperatureC: newStudy.temperatureC,
   117→   117→          humidityPercent: newStudy.humidityPercent,
   118→   118→          durationMonths: newStudy.durationMonths,
   119→   119→          ph: newStudy.ph,
   120→   120→        }),
   121→   121→      })
   122→   122→      if (res.ok) {
   123→   123→        toast({
   124→   124→          title: 'Study created',
   125→   125→          description: `${newStudy.substanceName} study created successfully (${code})`,
   126→   126→        })
   127→   127→        setCreateOpen(false)
   128→   128→        setNewStudy({ substanceName: '', studyType: 'long_term', temperatureC: 25, humidityPercent: 60, durationMonths: 24, ph: 7.0 })
   129→   129→        handleRefresh()
   130→   130→      } else {
   131→   131→        const errBody = await res.json().catch(() => ({}))
   132→   132→        toast({ title: 'Error', description: errBody.error || 'Failed to create study', variant: 'destructive' })
   133→   133→      }
   134→   134→    } catch {
   135→   135→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   136→   136→    } finally {
   137→   137→      setCreating(false)
   138→   138→    }
   139→   139→  }
   140→   140→
   141→   141→  const openDetail = async (study: StudyData) => {
   142→   142→    setDetailOpen(true)
   143→   143→    setDetailLoading(true)
   144→   144→    setDetailStudy(null)
   145→   145→    try {
   146→   146→      const res = await fetch(`/api/studies/${study.id}`)
   147→   147→      if (res.ok) {
   148→   148→        const data = await res.json()
   149→   149→        if (data.study) {
   150→   150→          setDetailStudy(data.study)
   151→   151→        } else {
   152→   152→          // Fallback: use the row data we already have
   153→   153→          setDetailStudy(study)
   154→   154→        }
   155→   155→      } else {
   156→   156→        setDetailStudy(study)
   157→   157→      }
   158→   158→    } catch {
   159→   159→      setDetailStudy(study)
   160→   160→    } finally {
   161→   161→      setDetailLoading(false)
   162→   162→    }
   163→   163→  }
   164→   164→
   165→   165→  const updateStudyStatus = async (newStatus: string) => {
   166→   166→    if (!detailStudy) return
   167→   167→    setStatusUpdating(true)
   168→   168→    try {
   169→   169→      const res = await fetch(`/api/studies/${detailStudy.id}`, {
   170→   170→        method: 'PUT',
   171→   171→        headers: { 'Content-Type': 'application/json' },
   172→   172→        body: JSON.stringify({
   173→   173→          status: newStatus,
   174→   174→          rejectionReason: newStatus === 'rejected' ? 'Rejected during review' : undefined,
   175→   175→        }),
   176→   176→      })
   177→   177→      if (res.ok) {
   178→   178→        const data = await res.json()
   179→   179→        if (data.study) {
   180→   180→          setDetailStudy((prev: any) => prev ? { ...prev, ...data.study } : prev)
   181→   181→        }
   182→   182→        toast({ title: 'Status updated', description: `Study marked as ${newStatus.replace('_', ' ')}` })
   183→   183→        handleRefresh()
   184→   184→      } else {
   185→   185→        toast({ title: 'Error', description: 'Failed to update status', variant: 'destructive' })
   186→   186→      }
   187→   187→    } catch {
   188→   188→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   189→   189→    } finally {
   190→   190→      setStatusUpdating(false)
   191→   191→    }
   192→   192→  }
   193→   193→
   194→   194→  const signStudy = async () => {
   195→   195→    if (!detailStudy) return
   196→   196→    setSigning(true)
   197→   197→    try {
   198→   198→      const res = await fetch(`/api/studies/${detailStudy.id}`, {
   199→   199→        method: 'POST',
   200→   200→        headers: { 'Content-Type': 'application/json' },
   201→   201→        body: JSON.stringify({
   202→   202→          signerName: 'Dr. Sarah Chen',
   203→   203→          signerRole: 'org_admin',
   204→   204→          meaning: 'Reviewed and Approved',
   205→   205→          newStatus: detailStudy.status === 'under_review' ? 'approved' : undefined,
   206→   206→        }),
   207→   207→      })
   208→   208→      if (res.ok) {
   209→   209→        const data = await res.json()
   210→   210→        // Refresh detail to show new signature
   211→   211→        if (data.study) {
   212→   212→          setDetailStudy((prev: any) => prev ? { ...prev, ...data.study, signatures: [...(prev.signatures || []), data.signature] } : prev)
   213→   213→        }
   214→   214→        toast({ title: 'Study signed', description: 'Electronic signature recorded (FDA 21 CFR Part 11)' })
   215→   215→        handleRefresh()
   216→   216→      } else {
   217→   217→        toast({ title: 'Error', description: 'Failed to sign study', variant: 'destructive' })
   218→   218→      }
   219→   219→    } catch {
   220→   220→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   221→   221→    } finally {
   222→   222→      setSigning(false)
   223→   223→    }
   224→   224→  }
   225→   225→
   226→   226→  const handleAddTimePoint = async () => {
   227→   227→    if (!detailStudy) return
   228→   228→    if (newTimePoint.timeDays < 0) {
   229→   229→      toast({ title: 'Validation error', description: 'Time (days) must be ≥ 0', variant: 'destructive' })
   230→   230→      return
   231→   231→    }
   232→   232→    setAddingTimePoint(true)
   233→   233→    try {
   234→   234→      const degradationPercent = Math.max(0, 100 - newTimePoint.percentRemaining)
   235→   235→      const res = await fetch('/api/timepoints', {
   236→   236→        method: 'POST',
   237→   237→        headers: { 'Content-Type': 'application/json' },
   238→   238→        body: JSON.stringify({
   239→   239→          timeDays: newTimePoint.timeDays,
   240→   240→          percentRemaining: newTimePoint.percentRemaining,
   241→   241→          degradationPercent,
   242→   242→          isOOS: newTimePoint.isOOS,
   243→   243→          isOOT: newTimePoint.isOOT,
   244→   244→          studyId: detailStudy.id,
   245→   245→        }),
   246→   246→      })
   247→   247→      if (res.ok) {
   248→   248→        const data = await res.json()
   249→   249→        toast({ title: 'Time point added', description: `Day ${newTimePoint.timeDays} recorded` })
   250→   250→        setDetailStudy((prev: any) => prev
   251→   251→          ? { ...prev, timePoints: [...(prev.timePoints || []), data.timePoint].sort((a: any, b: any) => a.timeDays - b.timeDays) }
   252→   252→          : prev)
   253→   253→        setNewTimePoint({ timeDays: 0, percentRemaining: 100, isOOS: false, isOOT: false })
   254→   254→      } else {
   255→   255→        const err = await res.json().catch(() => ({}))
   256→   256→        toast({ title: 'Error', description: err.error || 'Failed to add time point', variant: 'destructive' })
   257→   257→      }
   258→   258→    } catch {
   259→   259→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   260→   260→    } finally {
   261→   261→      setAddingTimePoint(false)
   262→   262→    }
   263→   263→  }
   264→   264→
   265→   265→  const handleDeleteTimePoint = async (tpId: string) => {
   266→   266→    setDeletingTimePointId(tpId)
   267→   267→    try {
   268→   268→      const res = await fetch(`/api/timepoints/${tpId}`, { method: 'DELETE' })
   269→   269→      if (res.ok) {
   270→   270→        toast({ title: 'Time point deleted', description: 'Measurement removed from study' })
   271→   271→        setDetailStudy((prev: any) => prev
   272→   272→          ? { ...prev, timePoints: (prev.timePoints || []).filter((tp: any) => tp.id !== tpId) }
   273→   273→          : prev)
   274→   274→      } else {
   275→   275→        toast({ title: 'Error', description: 'Failed to delete time point', variant: 'destructive' })
   276→   276→      }
   277→   277→    } catch {
   278→   278→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   279→   279→    } finally {
   280→   280→      setDeletingTimePointId(null)
   281→   281→    }
   282→   282→  }
   283→   283→
   284→   284→  const exportStudiesCSV = async () => {
   285→   285→    setExporting(true)
   286→   286→    try {
   287→   287→      const res = await fetch('/api/studies?limit=1000')
   288→   288→      if (res.ok) {
   289→   289→        const data = await res.json()
   290→   290→        const rows = (data.studies || []).map((s: any) => ({
   291→   291→          id: s.id,
   292→   292→          studyCode: s.studyCode ?? '',
   293→   293→          substanceName: s.substanceName ?? '',
   294→   294→          studyType: s.studyType ?? '',
   295→   295→          status: s.status ?? '',
   296→   296→          temperatureC: s.temperatureC ?? '',
   297→   297→          humidityPercent: s.humidityPercent ?? '',
   298→   298→          durationMonths: s.durationMonths ?? '',
   299→   299→          predictedShelfLifeMonths: s.predictedShelfLifeMonths ?? '',
   300→   300→          ph: s.ph ?? '',
   301→   301→        }))
   302→   302→        if (!rows.length) {
   303→   303→          toast({ title: 'Nothing to export', description: 'No studies found to export' })
   304→   304→        } else {
   305→   305→          exportCSV(rows, `chemstab-studies-${new Date().toISOString().slice(0, 10)}.csv`)
   306→   306→          toast({ title: 'Export complete', description: `Exported ${rows.length} studies to CSV` })
   307→   307→        }
   308→   308→      } else {
   309→   309→        toast({ title: 'Export failed', description: 'Failed to fetch studies', variant: 'destructive' })
   310→   310→      }
   311→   311→    } catch {
   312→   312→      toast({ title: 'Export failed', description: 'Network error', variant: 'destructive' })
   313→   313→    } finally {
   314→   314→      setExporting(false)
   315→   315→    }
   316→   316→  }
   317→   317→
   318→   318→  return (
   319→   319→    <motion.div
   320→   320→      initial={{ opacity: 0, y: 20 }}
   321→   321→      animate={{ opacity: 1, y: 0 }}
   322→   322→      transition={{ duration: 0.4 }}
   323→   323→      className="space-y-6"
   324→   324→    >
   325→   325→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   326→   326→        <div>
   327→   327→          <h1 className="text-2xl font-bold tracking-tight bg-gradient-to-r from-emerald-600 to-teal-600 dark:from-emerald-400 dark:to-teal-400 bg-clip-text text-transparent">Studies Management</h1>
   328→   328→          <p className="text-muted-foreground">Manage and track stability studies across your organization</p>
   329→   329→        </div>
   330→   330→        <div className="flex gap-2 flex-wrap">
   331→   331→          <Button variant="outline" size="sm" onClick={exportStudiesCSV} disabled={exporting}>
   332→   332→            <Download className={`size-4 mr-1 ${exporting ? 'animate-pulse' : ''}`} /> Export CSV
   333→   333→          </Button>
   334→   334→          <Button variant="outline" size="sm" onClick={handleRefresh} disabled={loading}>
   335→   335→            <RefreshCw className={`size-4 mr-1 ${loading ? 'animate-spin' : ''}`} /> Refresh
   336→   336→          </Button>
   337→   337→          <Button className="bg-emerald-600 hover:bg-emerald-700 text-white" onClick={() => setCreateOpen(true)}>
   338→   338→            <Plus className="size-4 mr-2" /> Create Study
   339→   339→          </Button>
   340→   340→        </div>
   341→   341→      </div>
   342→   342→
   343→   343→      {/* Status Pipeline Visualization */}
   344→   344→      <div className="flex items-center gap-1 p-3 rounded-xl bg-gradient-to-r from-emerald-50/50 to-teal-50/50 dark:from-emerald-950/20 dark:to-teal-950/20 border">
   345→   345→        {['draft', 'in_progress', 'under_review', 'approved'].map((stage, i) => {
   346→   346→          const count = studiesByStatus?.find(s => s.status === stage)?._count?.status || 0
   347→   347→          const stageColors: Record<string, string> = {
   348→   348→            draft: 'bg-slate-400', in_progress: 'bg-teal-500', under_review: 'bg-amber-500', approved: 'bg-emerald-500'
   349→   349→          }
   350→   350→          const stageLabels: Record<string, string> = {
   351→   351→            draft: 'Draft', in_progress: 'In Progress', under_review: 'Under Review', approved: 'Approved'
   352→   352→          }
   353→   353→          return (
   354→   354→            <div key={stage} className="flex items-center gap-1">
   355→   355→              {i > 0 && <ArrowRight className="size-3 text-muted-foreground" />}
   356→   356→              <div className="flex items-center gap-1.5 px-2 py-1 rounded-full bg-white/60 dark:bg-card/60">
   357→   357→                <span className={`size-2 rounded-full ${stageColors[stage]}`} />
   358→   358→                <span className="text-xs font-medium">{stageLabels[stage]}</span>
   359→   359→                <Badge variant="secondary" className="text-[10px] h-4 px-1">{count}</Badge>
   360→   360→              </div>
   361→   361→            </div>
   362→   362→          )
   363→   363→        })}
   364→   364→      </div>
   365→   365→
   366→   366→      {/* Study Type Filter Pills */}
   367→   367→      <div className="flex flex-wrap gap-2 mb-1">
   368→   368→        {['all', 'long_term', 'accelerated', 'intermediate', 'stress'].map((type) => (
   369→   369→          <Button
   370→   370→            key={type}
   371→   371→            variant={studyTypeFilter === type ? 'default' : 'outline'}
   372→   372→            size="sm"
   373→   373→            className={`h-7 text-xs rounded-full ${studyTypeFilter === type ? 'bg-emerald-600 text-white hover:bg-emerald-700' : ''}`}
   374→   374→            onClick={() => setStudyTypeFilter(type)}
   375→   375→          >
   376→   376→            {type === 'all' ? 'All Types' : studyTypeLabels[type]}
   377→   377→          </Button>
   378→   378→        ))}
   379→   379→      </div>
   380→   380→
   381→   381→      {/* Status Filter */}
   382→   382→      <div className="flex gap-3">
   383→   383→        <Select value={statusFilter} onValueChange={setStatusFilter}>
   384→   384→          <SelectTrigger className="w-[160px]">
   385→   385→            <SelectValue placeholder="Filter by Status" />
   386→   386→          </SelectTrigger>
   387→   387→          <SelectContent>
   388→   388→            <SelectItem value="all">All Statuses</SelectItem>
   389→   389→            <SelectItem value="draft">Draft</SelectItem>
   390→   390→            <SelectItem value="in_progress">In Progress</SelectItem>
   391→   391→            <SelectItem value="completed">Completed</SelectItem>
   392→   392→            <SelectItem value="under_review">Under Review</SelectItem>
   393→   393→            <SelectItem value="approved">Approved</SelectItem>
   394→   394→          </SelectContent>
   395→   395→        </Select>
   396→   396→      </div>
   397→   397→
   398→   398→      {/* Studies Table */}
   399→   399→      <Card className="backdrop-blur-sm bg-card/80">
   400→   400→        <CardContent className="p-0">
   401→   401→          <div className="max-h-96 overflow-y-auto">
   402→   402→            {loading ? (
   403→   403→              <div className="space-y-2 p-4">
   404→   404→                {Array.from({ length: 5 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   405→   405→              </div>
   406→   406→            ) : (
   407→   407→              <Table>
   408→   408→                <TableHeader>
   409→   409→                  <TableRow>
   410→   410→                    <TableHead>Study Code</TableHead>
   411→   411→                    <TableHead>Substance</TableHead>
   412→   412→                    <TableHead>Type</TableHead>
   413→   413→                    <TableHead>Temp (°C)</TableHead>
   414→   414→                    <TableHead>Shelf Life</TableHead>
   415→   415→                    <TableHead>Status</TableHead>
   416→   416→                  </TableRow>
   417→   417→                </TableHeader>
   418→   418→                <TableBody>
   419→   419→                  {apiStudies.map((std, idx) => (
   420→   420→                    <TableRow
   421→   421→                      key={std.id}
   422→   422→                      className={`cursor-pointer hover:bg-emerald-50/50 dark:hover:bg-emerald-900/10 transition-colors ${idx % 2 === 1 ? 'bg-muted/30' : ''}`}
   423→   423→                      onClick={() => openDetail(std)}
   424→   424→                    >
   425→   425→                      <TableCell className="font-medium font-mono">{std.studyCode}</TableCell>
   426→   426→                      <TableCell>{std.substanceName}</TableCell>
   427→   427→                      <TableCell>{studyTypeLabels[std.studyType]}</TableCell>
