   236→export function exportCSV(data: Record<string, unknown>[], filename: string) {
   237→  if (!data.length) return
   238→  const headers = Object.keys(data[0])
   239→  const csvLines = [headers.join(',')]
   240→  for (const row of data) {
   241→    const values = headers.map((h) => {
   242→      const v = row[h] ?? ''
   243→      const s = String(v).replace(/"/g, '""')
   244→      return /[",\n]/.test(s) ? `"${s}"` : s
   245→    })
   246→    csvLines.push(values.join(','))
   247→  }
   248→  const blob = new Blob([csvLines.join('\n')], { type: 'text/csv;charset=utf-8;' })
   249→  const url = URL.createObjectURL(blob)
   250→  const a = document.createElement('a')
   251→  a.href = url
   252→  a.download = filename
   253→  document.body.appendChild(a)
   254→  a.click()
   255→  document.body.removeChild(a)
   256→  URL.revokeObjectURL(url)
   257→}

   259→// Format a number for CSV output (null -> empty string)
   260→export function fmtNum(v: number | null | undefined): string {
   261→  return v === null || v === undefined ? '' : String(v)
   262→}

   264→// ── Chemical formula subscript formatter ──────────────────────────────
   265→// Converts "C9H8O4" → "C₉H₈O₄", "H2O2" → "H₂O₂", "NaCl" stays "NaCl"
   266→export const SUBSCRIPT_DIGITS: Record<string, string> = {
   267→  '0': '₀', '1': '₁', '2': '₂', '3': '₃', '4': '₄',
   268→  '5': '₅', '6': '₆', '7': '₇', '8': '₈', '9': '₉',
   269→}
   270→export function formatFormula(formula: string | null | undefined): string {
   271→  if (!formula) return '—'
   272→  // If already contains unicode subscripts, leave as-is
   273→  if (/[₀-₉]/.test(formula)) return formula
   274→  // Replace any digit that follows a letter (or another digit) with subscript
   275→  return formula.replace(/(\d+)/g, (m) =>
   276→    m.split('').map((d) => SUBSCRIPT_DIGITS[d] ?? d).join('')
   277→  )
   278→}

   323→// Helper: format an ISO timestamp into a relative-time string ("2h ago", "Yesterday", "3d ago")
   324→export function formatRelativeTime(timestamp: string): string {
   325→  const now = Date.now()
   326→  const then = new Date(timestamp).getTime()
   327→  if (Number.isNaN(then)) return ''
   328→  const diffMs = now - then
   329→  const sec = Math.max(0, Math.floor(diffMs / 1000))
   330→  if (sec < 60) return 'just now'
   331→  const min = Math.floor(sec / 60)
   332→  if (min < 60) return `${min}m ago`
   333→  const hr = Math.floor(min / 60)
   334→  if (hr < 24) return `${hr}h ago`
   335→  const day = Math.floor(hr / 24)
   336→  if (day === 1) return 'Yesterday'
   337→  if (day < 7) return `${day}d ago`
   338→  const wk = Math.floor(day / 7)
   339→  if (wk < 5) return `${wk}w ago`
   340→  const mo = Math.floor(day / 30)
   341→  if (mo < 12) return `${mo}mo ago`
   342→  return `${Math.floor(day / 365)}y ago`
   343→}
