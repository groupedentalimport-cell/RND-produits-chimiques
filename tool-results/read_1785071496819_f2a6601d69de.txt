    65→  const [degradationLoading, setDegradationLoading] = useState(false)
    66→  const [newDegradation, setNewDegradation] = useState({ name: '', smiles: '', percentage: 0, hazardLevel: 'low' })

    70→  const [viewMode, setViewMode] = useState<'table' | 'grid'>('table')
    71→  const fileInputRef = useRef<HTMLInputElement | null>(null)
    72→  const [newMolecule, setNewMolecule] = useState({
    73→    name: '',
    74→    casNumber: '',
    75→    smiles: '',
    76→    formula: '',
    77→    molarMass: 0,

    86→  useEffect(() => {
    87→    let cancelled = false

    98→          const transformed: MoleculeData[] = (data.molecules || []).map(transformMolecule)

   100→            setApiMolecules(transformed)
   101→            setTotalPages(data.pagination?.totalPages ?? 1)

   106→      if (!cancelled) setLoading(false)

   110→  }, [searchQuery, riskFilter, currentPageNum, refreshKey])
   111→
   112→  // Apply source filter client-side on fetched molecules, then sort
   113→  const displayed = (() => {
   114→    let filtered = sourceFilter === 'all'
   115→      ? apiMolecules
   116→      : apiMolecules.filter((mol) => mol.dataSource.toLowerCase() === sourceFilter.toLowerCase())
   117→    // Sort
   118→    if (sortBy === 'name') filtered.sort((a, b) => a.name.localeCompare(b.name))
   119→    else if (sortBy === 'stability') filtered.sort((a, b) => b.stabilityScore - a.stabilityScore)
   120→    else if (sortBy === 'molarMass') filtered.sort((a, b) => b.molarMass - a.molarMass)
   121→    else if (sortBy === 'risk') {
   122→      const riskOrder: Record<string, number> = { critical: 0, high: 1, moderate: 2, low: 3 }
   123→      filtered.sort((a, b) => (riskOrder[a.riskLevel] ?? 4) - (riskOrder[b.riskLevel] ?? 4))
   124→    }
   125→    return filtered
   126→  })()
   127→
   128→  // Count per risk level (from apiMolecules for filter pills)
   129→  const riskCounts = (() => {
   130→    const counts: Record<string, number> = { low: 0, moderate: 0, high: 0, critical: 0 }
   131→    apiMolecules.forEach((mol) => { counts[mol.riskLevel] = (counts[mol.riskLevel] || 0) + 1 })
   132→    return counts
   133→  })()
   134→
   135→  const openDetail = async (mol: MoleculeData) => {
   136→    setSelectedMolecule(mol)
   137→    setDetailOpen(true)
   138→    setDegradationProducts([])
   139→    setDegradationLoading(true)
   140→    try {
   141→      const res = await fetch(`/api/degradation-products?moleculeId=${mol.id}`)
   142→      if (res.ok) {
   143→        const data = await res.json()
   144→        setDegradationProducts(data.products || [])
   145→      }
   146→    } catch { /* ignore */ }
   147→    setDegradationLoading(false)
   148→  }
   149→
   150→  const handleAddDegradation = async () => {
   151→    if (!selectedMolecule || !newDegradation.name.trim()) {
   152→      toast({ title: 'Validation error', description: 'Product name is required', variant: 'destructive' })
   153→      return
   154→    }
   155→    setAddingDegradation(true)
   156→    try {
   157→      const res = await fetch('/api/degradation-products', {
   158→        method: 'POST',
   159→        headers: { 'Content-Type': 'application/json' },
   160→        body: JSON.stringify({
   161→          name: newDegradation.name,
   162→          smiles: newDegradation.smiles || undefined,

   166→        }),
   167→      })
   168→      if (res.ok) {
   169→        toast({ title: 'Degradation product added', description: `${newDegradation.name} linked to ${selectedMolecule.name}` })
   170→        setNewDegradation({ name: '', smiles: '', percentage: 0, hazardLevel: 'low' })
   171→        // Refresh list
   172→        const refreshRes = await fetch(`/api/degradation-products?moleculeId=${selectedMolecule.id}`)
   173→        if (refreshRes.ok) {
   174→          const data = await refreshRes.json()
   175→          setDegradationProducts(data.products || [])
   176→        }
   177→      } else {
   178→        const err = await res.json().catch(() => ({}))
   179→        toast({ title: 'Error', description: err.error || 'Failed to add degradation product', variant: 'destructive' })
   180→      }
   181→    } catch {
   182→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   183→    } finally {
   184→      setAddingDegradation(false)
   185→    }
   186→  }
   187→
   188→  const handleImportCSV = async (e: React.ChangeEvent<HTMLInputElement>) => {
   189→    const file = e.target.files?.[0]
   190→    if (!file) return
   191→    setImporting(true)
   192→    try {
   193→      const text = await file.text()
   194→      const lines = text.split(/\r?\n/).filter((l) => l.trim())
   195→      if (lines.length < 2) {
   196→        toast({ title: 'Import failed', description: 'CSV must have a header row and at least one data row', variant: 'destructive' })
   197→        return
   198→      }
   199→      const parseCSVLine = (line: string): string[] => {
   200→        const out: string[] = []
   201→        let cur = ''
   202→        let inQ = false
   203→        for (let i = 0; i < line.length; i++) {
   204→          const ch = line[i]
   205→          if (ch === '"' && line[i + 1] === '"') { cur += '"'; i++ }
   206→          else if (ch === '"') { inQ = !inQ }
   207→          else if (ch === ',' && !inQ) { out.push(cur); cur = '' }
   208→          else { cur += ch }
   209→        }
   210→        out.push(cur)
   211→        return out
   212→      }
   213→      const headers = parseCSVLine(lines[0]).map((h) => h.trim().toLowerCase())
   214→      const nameIdx = headers.findIndex((h) => h === 'name')
   215→      if (nameIdx === -1) {
   216→        toast({ title: 'Import failed', description: 'CSV must have a "name" column', variant: 'destructive' })
   217→        return
   218→      }
   219→      const casIdx = headers.findIndex((h) => h === 'casnumber' || h === 'cas')
   220→      const formulaIdx = headers.findIndex((h) => h === 'formula')
   221→      const smilesIdx = headers.findIndex((h) => h === 'smiles')
   222→      const mwIdx = headers.findIndex((h) => h === 'molarmass' || h === 'mw')
   223→      const logPIdx = headers.findIndex((h) => h === 'logp')
   224→      const scoreIdx = headers.findIndex((h) => h === 'stabilityscore' || h === 'predictedstabilityscore' || h === 'score')
   225→      const riskIdx = headers.findIndex((h) => h === 'risklevel' || h === 'risk')
   226→
   227→      let success = 0, failed = 0
   228→      for (let i = 1; i < lines.length; i++) {
   229→        const cells = parseCSVLine(lines[i])
   230→        const name = cells[nameIdx]?.trim()
   231→        if (!name) { failed++; continue }
   232→        try {
   233→          const res = await fetch('/api/molecules', {
   234→            method: 'POST',
   235→            headers: { 'Content-Type': 'application/json' },
   236→            body: JSON.stringify({
   237→              name,
   238→              casNumber: casIdx >= 0 ? cells[casIdx]?.trim() || undefined : undefined,
   239→              formula: formulaIdx >= 0 ? cells[formulaIdx]?.trim() || undefined : undefined,
   240→              smiles: smilesIdx >= 0 ? cells[smilesIdx]?.trim() || undefined : undefined,
   241→              molarMass: mwIdx >= 0 ? parseFloat(cells[mwIdx]) || undefined : undefined,
   242→              logP: logPIdx >= 0 ? parseFloat(cells[logPIdx]) || undefined : undefined,
   243→              predictedStabilityScore: scoreIdx >= 0 ? parseFloat(cells[scoreIdx]) || undefined : undefined,
   244→              riskLevel: riskIdx >= 0 ? cells[riskIdx]?.trim().toLowerCase() || 'low' : 'low',
   245→            }),
   246→          })
   247→          if (res.ok) success++; else failed++
   248→        } catch { failed++ }
   249→      }
   250→      toast({
   251→        title: 'Import complete',
   252→        description: `Imported ${success} molecule${success !== 1 ? 's' : ''}${failed ? `, ${failed} failed` : ''}`,
   253→        variant: failed > 0 ? 'destructive' : 'default',
   254→      })
   255→      handleRefresh()
   256→    } catch {
   257→      toast({ title: 'Import failed', description: 'Failed to read file', variant: 'destructive' })
   258→    } finally {
   259→      setImporting(false)
   260→      if (fileInputRef.current) fileInputRef.current.value = ''
   261→    }
   262→  }
   263→
   264→  const handleRefresh = () => {
   265→    setLoading(true)
   266→    setRefreshKey(k => k + 1)
   267→  }
   268→
   269→  const resetNewMolecule = () => {

   273→    })
   274→  }
   275→
   276→  const handleAddMolecule = async () => {

   278→      toast({ title: 'Validation error', description: 'Molecule name is required', variant: 'destructive' })
   279→      return
   280→    }

   282→    try {
   283→      const res = await fetch('/api/molecules', {
   284→        method: 'POST',
   285→        headers: { 'Content-Type': 'application/json' },
   286→        body: JSON.stringify({
   287→          name: newMolecule.name,
   288→          casNumber: newMolecule.casNumber || undefined,
   289→          smiles: newMolecule.smiles || undefined,
   290→          formula: newMolecule.formula || undefined,
   291→          molarMass: newMolecule.molarMass || undefined,
   292→          logP: newMolecule.logP || undefined,
   293→          predictedStabilityScore: newMolecule.predictedStabilityScore || undefined,
   294→          riskLevel: newMolecule.riskLevel,
   295→          dataSource: newMolecule.dataSource,
   296→          description: newMolecule.description || undefined,
   297→        }),
   298→      })
   299→      if (res.ok) {
   300→        toast({ title: 'Molecule added', description: `${newMolecule.name} created successfully` })
   301→        setAddOpen(false)
   302→        resetNewMolecule()
   303→        handleRefresh()
   304→      } else {
   305→        const errBody = await res.json().catch(() => ({}))
   306→        toast({ title: 'Error', description: errBody.error || 'Failed to create molecule', variant: 'destructive' })
   307→      }
   308→    } catch {
   309→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   310→    } finally {
   311→      setCreating(false)
   312→    }
   313→  }
   314→
   315→  const exportMoleculesCSV = async () => {
   316→    setExporting(true)
   317→    try {
   318→      const res = await fetch('/api/molecules?limit=1000')
   319→      if (res.ok) {
   320→        const data = await res.json()
   321→        const rows = (data.molecules || []).map((m: any) => ({
   322→          id: m.id,
   323→          name: m.name,
   324→          casNumber: m.casNumber ?? '',
   325→          smiles: m.smiles ?? '',
   326→          formula: m.formula ?? '',
   327→          molarMass: m.molarMass ?? '',
   328→          logP: m.logP ?? '',
   329→          stabilityScore: m.predictedStabilityScore ?? '',
   330→          riskLevel: m.riskLevel ?? '',
   331→          dataSource: m.dataSource ?? '',
   332→          meltingPoint: m.meltingPoint ?? '',
   333→          boilingPoint: m.boilingPoint ?? '',
   334→          description: m.description ?? '',
   335→        }))
   336→        if (!rows.length) {
   337→          toast({ title: 'Nothing to export', description: 'No molecules found to export' })
   338→        } else {
   339→          exportCSV(rows, `chemstab-molecules-${new Date().toISOString().slice(0, 10)}.csv`)
   340→          toast({ title: 'Export complete', description: `Exported ${rows.length} molecules to CSV` })
   341→        }
   342→      } else {
   343→        toast({ title: 'Export failed', description: 'Failed to fetch molecules', variant: 'destructive' })
   344→      }
   345→    } catch {
   346→      toast({ title: 'Export failed', description: 'Network error', variant: 'destructive' })
   347→    } finally {
   348→      setExporting(false)
   349→    }
   350→  }

   352→  return (
   353→    <motion.div
   354→      initial={{ opacity: 0, y: 20 }}
   355→      animate={{ opacity: 1, y: 0 }}
   356→      transition={{ duration: 0.4 }}
   357→      className="space-y-6"
   358→    >
   359→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   360→        <div>
   361→          <h1 className="text-2xl font-bold tracking-tight bg-gradient-to-r from-emerald-600 to-teal-600 dark:from-emerald-400 dark:to-teal-400 bg-clip-text text-transparent">Molecules Database</h1>
   362→          <p className="text-muted-foreground">Browse and search chemical compounds with stability assessments</p>
   363→        </div>
   364→        <div className="flex gap-2 flex-wrap items-center">
   365→          {/* View Mode Toggle */}
   366→          <div className="flex gap-1 mr-1">

   371→              <LayoutGrid className="size-4" /> Grid
   372→            </Button>
   373→          </div>
   374→          <input
   375→            type="file"
   376→            accept=".csv"
   377→            ref={fileInputRef}
   378→            onChange={handleImportCSV}
   379→            className="hidden"
   380→          />
   381→          <TooltipProvider delayDuration={200}>

   383→              <TooltipTrigger asChild>

   385→                  <Download className={`size-4 mr-1 rotate-180 ${importing ? 'animate-pulse' : ''}`} /> Import CSV

   387→              </TooltipTrigger>
   388→              <TooltipContent>Bulk import molecules from a CSV file (name column required)</TooltipContent>

   390→          </TooltipProvider>

   400→          </TooltipProvider>

   402→            <RefreshCw className={`size-4 mr-1 ${loading ? 'animate-spin' : ''}`} /> Refresh

   407→        </div>

   412→        {['all', 'low', 'moderate', 'high', 'critical'].map((level) => (
   413→          <Button
   414→            key={level}
   415→            variant={riskFilter === level ? 'default' : 'outline'}
   416→            size="sm"
   417→            className={`h-7 text-xs rounded-full ${riskFilter === level ? (RISK_PILL_ACTIVE[level] || 'bg-emerald-600 text-white') : ''} ${RISK_PILL_OUTLINE[level] || ''}`}
   418→            onClick={() => { setRiskFilter(level); setCurrentPageNum(1) }}
   419→          >
   420→            {level === 'all' ? 'All' : level.charAt(0).toUpperCase() + level.slice(1)}
   421→            {level !== 'all' && <span className="ml-1 opacity-60">({riskCounts[level] || 0})</span>}
   422→          </Button>
   423→        ))}
   424→      </div>
   425→
   426→      {/* Search & Filters */}
   427→      <div className="flex flex-col sm:flex-row gap-3">
   428→        <div className="relative flex-1">
   429→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   430→          <Input
   431→            placeholder="Search by name, CAS, SMILES..."
   432→            value={searchQuery}
   433→            onChange={(e) => { setSearchQuery(e.target.value); setCurrentPageNum(1) }}
   434→            className="pl-9"
   435→          />
   436→        </div>
   437→        <Select value={sortBy} onValueChange={setSortBy}>
   438→          <SelectTrigger className="w-[130px]">
   439→            <SelectValue placeholder="Sort by" />
   440→          </SelectTrigger>
   441→          <SelectContent>
   442→            <SelectItem value="name">Name</SelectItem>
   443→            <SelectItem value="stability">Stability</SelectItem>
   444→            <SelectItem value="molarMass">Molar Mass</SelectItem>
   445→            <SelectItem value="risk">Risk Level</SelectItem>
   446→          </SelectContent>
   447→        </Select>
   448→        <Select value={riskFilter} onValueChange={(v) => { setRiskFilter(v); setCurrentPageNum(1) }}>
   449→          <SelectTrigger className="w-[140px]">
   450→            <SelectValue placeholder="Risk Level" />
   451→          </SelectTrigger>
   452→          <SelectContent>
   453→            <SelectItem value="all">All Risks</SelectItem>
   454→            <SelectItem value="low">Low</SelectItem>
   455→            <SelectItem value="moderate">Moderate</SelectItem>
   456→            <SelectItem value="high">High</SelectItem>
   457→            <SelectItem value="critical">Critical</SelectItem>
   458→          </SelectContent>
   459→        </Select>
   460→        <Select value={sourceFilter} onValueChange={(v) => { setSourceFilter(v); setCurrentPageNum(1) }}>
   461→          <SelectTrigger className="w-[140px]">
   462→            <SelectValue placeholder="Data Source" />
   463→          </SelectTrigger>
   464→          <SelectContent>
   465→            <SelectItem value="all">All Sources</SelectItem>
   466→            <SelectItem value="PubChem">PubChem</SelectItem>
   467→            <SelectItem value="ChEMBL">ChEMBL</SelectItem>
   468→            <SelectItem value="Manual">Manual</SelectItem>
   469→          </SelectContent>
   470→        </Select>
   471→      </div>
   472→
   473→      {/* Results count */}
   474→      <p className="text-sm text-muted-foreground">
   475→        {loading ? 'Loading...' : `Showing ${displayed.length} of ${totalCount} molecules`}
   476→      </p>
   477→
   478→      {/* Table / Grid View */}
   479→      {viewMode === 'table' ? (
   480→      <Card className="backdrop-blur-sm bg-card/80">
   481→        <CardContent className="p-0">
   482→          <div className="max-h-96 overflow-y-auto">
   483→            {loading ? (
   484→              <div className="space-y-2 p-4">
   485→                {Array.from({ length: 5 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   486→              </div>
   487→            ) : (
