   329→                    {users.length === 0 && (

   334→                  </TableBody>

   338→          </CardContent>

   340→
   341→        {/* Audit Trail with filters */}

   344→            <CardTitle className="flex items-center gap-2">
   345→              <ClipboardList className="size-5" />
   346→              Audit Trail

   349→          </CardHeader>
   350→          <CardContent className="space-y-3">

   355→                value={auditSearch}

   381→
   382→            {/* Action facet chips */}

   400→
   401→            {/* Audit Timeline */}
   402→            <div className="max-h-72 overflow-y-auto">
   403→              {auditLoading ? (
   404→                <div className="space-y-4 p-2">
   405→                  {Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-16 w-full" />)}
   406→                </div>
   407→              ) : auditData.length === 0 ? (
   408→                <p className="text-center text-muted-foreground py-10 text-xs">
   409→                  No audit events match your filters
   410→                </p>
   411→              ) : (
   412→                <div className="space-y-0">
   413→                  {auditData.map((entry, i) => {
   414→                    const Icon = ACTION_ICON_MAP[entry.action] || Activity
   415→                    const color = ACTION_COLOR_MAP[entry.action] || ACTION_COLOR_MAP.create
   416→                    return (
   417→                      <motion.div
   418→                        key={entry.id}
   419→                        initial={{ opacity: 0, x: -12 }}

   422→                        className="flex gap-4 pb-6 relative"
   423→                      >

   446→
   447→      {/* System Health Dashboard + System Configuration */}
   448→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   449→        {/* System Health Dashboard */}
   450→        <Card className="gradient-border backdrop-blur-sm bg-card/80">

   452→            <CardTitle className="flex items-center gap-2">

   457→          <CardContent className="space-y-4">
