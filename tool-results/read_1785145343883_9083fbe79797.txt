   238→export function exportCSV(data: Record<string, unknown>[], filename: string) {
   239→  if (!data.length) return
   240→  const headers = Object.keys(data[0])
   241→  const csvLines = [headers.join(',')]
   242→  for (const row of data) {
   243→    const values = headers.map((h) => {
   244→      const v = row[h] ?? ''
   245→      const s = String(v).replace(/"/g, '""')
   246→      return /[",\n]/.test(s) ? `"${s}"` : s
   247→    })
   248→    csvLines.push(values.join(','))
   249→  }
   250→  const blob = new Blob([csvLines.join('\n')], { type: 'text/csv;charset=utf-8;' })
   251→  const url = URL.createObjectURL(blob)
   252→  const a = document.createElement('a')
   253→  a.href = url
   254→  a.download = filename
   255→  document.body.appendChild(a)
   256→  a.click()
   257→  document.body.removeChild(a)
   258→  URL.revokeObjectURL(url)
   259→}

   266→// ── Chemical formula subscript formatter ──────────────────────────────
   267→// Converts "C9H8O4" → "C₉H₈O₄", "H2O2" → "H₂O₂", "NaCl" stays "NaCl"
   268→export const SUBSCRIPT_DIGITS: Record<string, string> = {
   269→  '0': '₀', '1': '₁', '2': '₂', '3': '₃', '4': '₄',
   270→  '5': '₅', '6': '₆', '7': '₇', '8': '₈', '9': '₉',
   271→}
   272→export function formatFormula(formula: string | null | undefined): string {
   273→  if (!formula) return '—'
   274→  // If already contains unicode subscripts, leave as-is
   275→  if (/[₀-₉]/.test(formula)) return formula
   276→  // Replace any digit that follows a letter (or another digit) with subscript
   277→  return formula.replace(/(\d+)/g, (m) =>
   278→    m.split('').map((d) => SUBSCRIPT_DIGITS[d] ?? d).join('')
   279→  )
   280→}

   325→// Helper: format an ISO timestamp into a relative-time string ("2h ago", "Yesterday", "3d ago")
   326→export function formatRelativeTime(timestamp: string): string {
   327→  const now = Date.now()
   328→  const then = new Date(timestamp).getTime()
   329→  if (Number.isNaN(then)) return ''
   330→  const diffMs = now - then
   331→  const sec = Math.max(0, Math.floor(diffMs / 1000))
   332→  if (sec < 60) return 'just now'
   333→  const min = Math.floor(sec / 60)
   334→  if (min < 60) return `${min}m ago`
   335→  const hr = Math.floor(min / 60)
   336→  if (hr < 24) return `${hr}h ago`
   337→  const day = Math.floor(hr / 24)
   338→  if (day === 1) return 'Yesterday'
   339→  if (day < 7) return `${day}d ago`
   340→  const wk = Math.floor(day / 7)
   341→  if (wk < 5) return `${wk}w ago`
   342→  const mo = Math.floor(day / 30)
   343→  if (mo < 12) return `${mo}mo ago`
   344→  return `${Math.floor(day / 365)}y ago`
   345→}
